# Map and Intermediate Camps Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:**
- Keokradong gets intermediate camps: Boga Lake → Camp 1 → Camp 2 → Summit.
- Every mountain gets an always-visible round minimap, with the player arrow turning to their movement direction.
- `Tab` opens a full map with camp names and elevations.

**Architecture:**
- The engine builds a shaded-relief map image once from the already-loaded low-res backdrop heights. It uses pure `mapImage.ts`, plus `mapMath.ts` for coordinates, route thinning, camp progress and heading.
- The engine publishes `map` once. It then publishes `mapPlayer`, `mapCamps` and `mapOpen` through the existing throttled HUD publish.
- React draws the minimap and the full map on `<canvas>` with a shared `drawMap` helper. The engine never touches the DOM for the map.

**Tech Stack:** TypeScript 5.9, Three.js engine (unchanged rendering), React 19 + Zustand HUD, Canvas 2D, Vitest (Node), Playwright, Python bake.

**Spec:** `docs/superpowers/specs/2026-11-07-map-design.md`.

## Global Constraints

- Work on branch `map`, created from `main`.
- `src/engine/**` never imports React. React (`src/ui`, `src/app`) never touches Three.js or Rapier objects. UI may import `src/engine/config/*` and types from `src/engine/core/hud.ts`.
- Every gameplay or tuning number lives in `GAME_CONFIG`. UI layout sizes (the minimap's pixel size, colours) are named constants in the UI files.
- HUD state is published at most every 100 ms, never per frame. The big `map` object is published once, after load.
- Map orientation is north-up. World +x is east and +z is south, so map `u = (x − minX) / sizeX` (left to right) and `v = (z − minZ) / sizeZ` (top to bottom).
- `headingRad` is 0 for north, increasing clockwise: π/2 east and ±π south.
- Keokradong camps are `boga-lake` 400, `camp-1` 580 and `camp-2` 800. `summitWalkable` stays true. Every route is below 33°, and every camp is at least 500 m from the grid edge.
- Show the DEM attribution verbatim: `Contains modified Copernicus DEM data`.
- Run `npm run format` before every commit. CI runs `format:check`, `lint`, `typecheck`, `test`, `build` and `e2e`. Python tests: `.venv/bin/python -m pytest tools/terrain -q`.
- Commit messages end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```ts
import type { TerrainManifest } from './terrainData';

type P3 = readonly [number, number, number];

/** World rectangle covered by the baked tiles (and the map image). */
export interface MapBounds {
  minX: number;
  minZ: number;
  sizeX: number;
  sizeZ: number;
}

export function mapBounds(m: TerrainManifest): MapBounds {
  const tileSize = (m.tileSamples - 1) * m.spacingM;
  return { minX: m.originX, minZ: m.originZ, sizeX: m.tilesX * tileSize, sizeZ: m.tilesZ * tileSize };
}

/** North-up map units: u grows east (0 = west edge), v grows south (0 = north edge). */
export const worldToMap = (x: number, z: number, b: MapBounds): { u: number; v: number } => ({
  u: (x - b.minX) / b.sizeX,
  v: (z - b.minZ) / b.sizeZ,
});

export const mapToWorld = (u: number, v: number, b: MapBounds): { x: number; z: number } => ({
  x: b.minX + u * b.sizeX,
  z: b.minZ + v * b.sizeZ,
});

/** Drops points closer than minSpacingM (horizontally) to the last kept one; keeps both ends. */
export function simplifyRoute(points: readonly P3[], minSpacingM: number): [number, number, number][] {
  if (points.length <= 2) return points.map((p) => [p[0], p[1], p[2]]);
  const out: [number, number, number][] = [[points[0][0], points[0][1], points[0][2]]];
  let last = points[0];
  for (let i = 1; i < points.length - 1; i++) {
    const p = points[i];
    if (Math.hypot(p[0] - last[0], p[2] - last[2]) >= minSpacingM) {
      out.push([p[0], p[1], p[2]]);
      last = p;
    }
  }
  const end = points[points.length - 1];
  out.push([end[0], end[1], end[2]]);
  return out;
}

export type CampState = 'reached' | 'next' | 'ahead';

/** Per camp: reached, the next one after the last checkpoint, or still ahead. */
export function campProgress(
  campIds: readonly string[],
  reached: ReadonlySet<string>,
  lastId: string,
): Record<string, CampState> {
  const lastIndex = campIds.indexOf(lastId);
  return Object.fromEntries(
    campIds.map((id, i): [string, CampState] => [
      id,
      reached.has(id) ? 'reached' : i === lastIndex + 1 ? 'next' : 'ahead',
    ]),
  );
}

/**
 * Map heading from PlayerSnapshot.facingYaw (atan2(facing.x, facing.z), world +z = south):
 * 0 = north, clockwise positive, normalised to [-π, π).
 */
export function mapHeading(facingYaw: number): number {
  const h = Math.PI - facingYaw;
  const twoPi = Math.PI * 2;
  return ((((h + Math.PI) % twoPi) + twoPi) % twoPi) - Math.PI;
}
```

- [ ] **Step 4: Implement `src/engine/world/mapImage.ts`**

```ts
import { linearPalette, terrainColor, type TerrainPalette } from './palette';
import type { SurfaceRules } from './surface';
import type { Backdrop } from './terrainData';

export interface MapImage {
  width: number;
  height: number;
  /** RGBA, row-major, row = z (north at the top). */
  pixels: Uint8ClampedArray;
}

/** Light from the north-west and above (cartographic hillshade convention). */
const LIGHT = (() => {
  const l = [-1, 1.4, -1];
  const len = Math.hypot(l[0], l[1], l[2]);
  return [l[0] / len, l[1] / len, l[2] / len] as const;
})();
const SHADE_BASE = 0.55;
const SHADE_GAIN = 0.45;
const SHADE_MIN = 0.35;

const linearToSrgb = (c: number): number =>
  c <= 0.0031308 ? 12.92 * c : 1.055 * c ** (1 / 2.4) - 0.055;

/** Shaded-relief map of the backdrop heights in the biome's terrain colours. */
export function buildMapImage(
  b: Backdrop,
  palette: TerrainPalette,
  surface: SurfaceRules,
): MapImage {
  const n = b.samples;
  const s = b.spacingM;
  const pal = linearPalette(palette);
  const pixels = new Uint8ClampedArray(n * n * 4);
  const h = (c: number, r: number): number =>
    b.heights[Math.min(Math.max(r, 0), n - 1) * n + Math.min(Math.max(c, 0), n - 1)];
  for (let r = 0; r < n; r++) {
    for (let c = 0; c < n; c++) {
      const dx = (h(c + 1, r) - h(c - 1, r)) / (2 * s);
      const dz = (h(c, r + 1) - h(c, r - 1)) / (2 * s);
      const len = Math.hypot(dx, 1, dz);
      const nx = -dx / len;
      const ny = 1 / len;
      const nz = -dz / len;
      const slopeDeg = (Math.acos(ny) * 180) / Math.PI;
      const color = terrainColor(slopeDeg, h(c, r), surface, pal);
      const lit = nx * LIGHT[0] + ny * LIGHT[1] + nz * LIGHT[2];
      const shade = Math.min(1, Math.max(SHADE_MIN, SHADE_BASE + SHADE_GAIN * lit));
      const o = (r * n + c) * 4;
      pixels[o] = Math.round(linearToSrgb(color[0] * shade) * 255);
      pixels[o + 1] = Math.round(linearToSrgb(color[1] * shade) * 255);
      pixels[o + 2] = Math.round(linearToSrgb(color[2] * shade) * 255);
      pixels[o + 3] = 255;
    }
  }
  return { width: n, height: n, pixels };
}
```

The shading constants are visual and local to the map image, the same as the tree shape constants.

import { useMemo } from 'react';
import type { MapInfo } from '../../engine/core/hud';

/** The relief image as an offscreen canvas, built once per map. */
export function useRelief(map: MapInfo | null): HTMLCanvasElement | null {
  return useMemo(() => {
    if (!map) return null;
    const { width, height, pixels } = map.image;
    const canvas = document.createElement('canvas');
    canvas.width = width;
    canvas.height = height;
    canvas
      .getContext('2d')
      ?.putImageData(new ImageData(new Uint8ClampedArray(pixels), width, height), 0, 0);
    return canvas;
  }, [map]);
}

- [ ] **Step 3: `src/ui/map/drawMap.ts`**

```ts
import type { HudState, MapInfo } from '../../engine/core/hud';

export interface MapView {
  /** Map units (0–1) at the canvas centre. */
  centerU: number;
  centerV: number;
  /** Map units across the canvas width. */
  spanU: number;
  /** Canvas size in device pixels (square). */
  sizePx: number;
  /** Device pixel ratio, for line widths and fonts. */
  dpr: number;
  labels: boolean;
  north: boolean;
}

const BACKGROUND = '#0d1117';
const ROUTE = '#ff8a1f';
const CAMP: Record<string, string> = { reached: '#6cc070', next: '#f2b84b', ahead: '#f5f7fa' };
const PLAYER = '#ff4d4d';

/** Draws relief, routes, camps, the player arrow and (optionally) labels and a north mark. */
export function drawMap(
  ctx: CanvasRenderingContext2D,
  relief: CanvasImageSource,
  map: MapInfo,
  player: HudState['mapPlayer'],
  camps: HudState['mapCamps'],
  view: MapView,
): void {
  const { sizePx: size, dpr } = view;
  const kx = size / view.spanU;
  const ky = (kx * map.sizeM.z) / map.sizeM.x;
  const px = (u: number) => (u - view.centerU) * kx + size / 2;
  const py = (v: number) => (v - view.centerV) * ky + size / 2;

  ctx.clearRect(0, 0, size, size);
  ctx.fillStyle = BACKGROUND;
  ctx.fillRect(0, 0, size, size);
  ctx.imageSmoothingEnabled = true;
  ctx.drawImage(relief, px(0), py(0), kx, ky);

  ctx.strokeStyle = ROUTE;
  ctx.lineWidth = (view.labels ? 3 : 2.5) * dpr;
  ctx.lineJoin = 'round';
  for (const route of map.routes) {
    ctx.beginPath();
    route.forEach((p, i) => (i === 0 ? ctx.moveTo(px(p.u), py(p.v)) : ctx.lineTo(px(p.u), py(p.v))));
    ctx.stroke();
  }

  ctx.font = `600 ${12 * dpr}px system-ui, sans-serif`;
  ctx.textBaseline = 'middle';
  for (const camp of map.camps) {
    const x = px(camp.u);
    const y = py(camp.v);
    ctx.beginPath();
    ctx.arc(x, y, 5 * dpr, 0, Math.PI * 2);
    ctx.fillStyle = CAMP[camps[camp.id] ?? 'ahead'];
    ctx.fill();
    ctx.lineWidth = 1.5 * dpr;
    ctx.strokeStyle = BACKGROUND;
    ctx.stroke();
    if (view.labels) {
      const text = `${camp.name} · ${camp.elevationM.toLocaleString('en-US')} m`;
      ctx.lineWidth = 3 * dpr;
      ctx.strokeText(text, x + 9 * dpr, y);
      ctx.fillStyle = '#ffffff';
      ctx.fillText(text, x + 9 * dpr, y);
    }
  }

  if (player) {
    ctx.save();
    ctx.translate(px(player.u), py(player.v));
    ctx.rotate(player.headingRad);
    ctx.beginPath();
    ctx.moveTo(0, -10 * dpr);
    ctx.lineTo(7 * dpr, 8 * dpr);
    ctx.lineTo(0, 4 * dpr);
    ctx.lineTo(-7 * dpr, 8 * dpr);
    ctx.closePath();
    ctx.fillStyle = PLAYER;
    ctx.fill();
    ctx.lineWidth = 1.5 * dpr;
    ctx.strokeStyle = '#ffffff';
    ctx.stroke();
    ctx.restore();
  }

  if (view.north) {
    ctx.font = `700 ${12 * dpr}px system-ui, sans-serif`;
    ctx.textAlign = 'center';
    ctx.lineWidth = 3 * dpr;
    ctx.strokeStyle = BACKGROUND;
    ctx.strokeText('N', size / 2, 12 * dpr);
    ctx.fillStyle = '#ffffff';
    ctx.fillText('N', size / 2, 12 * dpr);
    ctx.textAlign = 'start';
  }
}
```

import { useEffect, useRef } from 'react';
import { useStore } from 'zustand';
import { GAME_CONFIG } from '../../engine/config/game.config';
import { gameStore } from '../../state/gameStore';
import { drawMap } from './drawMap';
import { useRelief } from './useRelief';

const SIZE_PX = 180;

/** Always-visible north-up minimap centred on the player. */
export function Minimap() {
  const map = useStore(gameStore, (s) => s.map);
  const player = useStore(gameStore, (s) => s.mapPlayer);
  const camps = useStore(gameStore, (s) => s.mapCamps);
  const relief = useRelief(map);
  const ref = useRef<HTMLCanvasElement>(null);
  const dpr = Math.min(window.devicePixelRatio || 1, 2);

  useEffect(() => {
    const ctx = ref.current?.getContext('2d');
    if (!ctx || !map || !relief || !player) return;
    drawMap(ctx, relief, map, player, camps, {
      centerU: player.u,
      centerV: player.v,
      spanU: GAME_CONFIG.map.minimapSpanM / map.sizeM.x,
      sizePx: SIZE_PX * dpr,
      dpr,
      labels: false,
      north: true,
    });
  }, [map, relief, player, camps, dpr]);

  if (!map) return null;
  return (
    <canvas
      ref={ref}
      className="minimap"
      role="img"
      aria-label="Map"
      width={SIZE_PX * dpr}
      height={SIZE_PX * dpr}
    />
  );
}

import { useEffect, useRef } from 'react';
import { useStore } from 'zustand';
import { gameStore } from '../../state/gameStore';
import { formatDistance } from '../formatDistance';
import { drawMap } from './drawMap';
import { useRelief } from './useRelief';

const SIZE_PX = 720;

/** Whole-mountain map, toggled with Tab. */
export function FullMap({ mountainName }: { mountainName: string }) {
  const open = useStore(gameStore, (s) => s.mapOpen);
  const map = useStore(gameStore, (s) => s.map);
  const player = useStore(gameStore, (s) => s.mapPlayer);
  const camps = useStore(gameStore, (s) => s.mapCamps);
  const guide = useStore(gameStore, (s) => s.guide);
  const relief = useRelief(map);
  const ref = useRef<HTMLCanvasElement>(null);
  const dpr = Math.min(window.devicePixelRatio || 1, 2);

  useEffect(() => {
    const ctx = ref.current?.getContext('2d');
    if (!open || !ctx || !map || !relief) return;
    drawMap(ctx, relief, map, player, camps, {
      centerU: 0.5,
      centerV: 0.5,
      spanU: 1,
      sizePx: SIZE_PX * dpr,
      dpr,
      labels: true,
      north: true,
    });
  }, [open, map, relief, player, camps, dpr]);

  if (!open || !map) return null;
  return (
    <div className="fullmap" role="dialog" aria-label={`Map of ${mountainName}`}>
      <div className="fullmap__title">{mountainName}</div>
      <canvas ref={ref} className="fullmap__canvas" width={SIZE_PX * dpr} height={SIZE_PX * dpr} />
      <div className="fullmap__footer">
        {guide ? `Next: ${guide.label} · ${formatDistance(guide.distanceM)}` : 'All camps reached'}
        {' · Tab — close'}
      </div>
      <ul className="sr-only">
        {map.camps.map((c) => (
          <li key={c.id}>{`${c.name} · ${c.elevationM} m · ${camps[c.id] ?? 'ahead'}`}</li>
        ))}
      </ul>
    </div>
  );
}
