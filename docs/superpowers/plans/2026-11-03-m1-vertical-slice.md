# Mountain Odyssey — Phase 0 + M1 Vertical Slice Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a playable browser vertical slice: a third-person character walks, runs, slides and climbs on streamed DEM-based terrain from Khare to High Camp, with stamina, fall damage and a checkpoint.

**Architecture:** Vite + React renders only the shell and HUD. A pure-TypeScript engine (`src/engine`) owns the game loop, Three.js rendering and Rapier physics, and publishes HUD state to a Zustand store at most every 100 ms. Terrain is baked offline by a Python pipeline into raw `uint16` tiles that the engine streams in chunks with LOD.

**Tech Stack:** Vite 8, TypeScript 5.9 (strict), React 19, Three.js 0.187, `@dimforge/rapier3d-compat` 0.21, Zustand 5, Vitest, Playwright, Python 3.9+ (numpy, Pillow, rasterio).

**Spec:** [docs/superpowers/specs/2026-11-03-mountain-odyssey-design.md](../specs/2026-11-03-mountain-odyssey-design.md) — this plan covers spec §12 "Phase 0" and "M1".

## Global Constraints

- Desktop browsers with WebGL2 only; keyboard + mouse input.
- TypeScript `strict: true`. No `any` in engine code.
- `src/engine/**` must never import React (enforced by ESLint in Task 1).
- React never touches Three.js or Rapier objects; it only calls the `Game` public API.
- Engine publishes HUD state at most every 100 ms, never per frame.
- Three.js renders only; Rapier simulates only.
- Simulation runs on a fixed 60 Hz step with at most 5 steps per frame.
- Every tunable gameplay number lives in `src/engine/config/game.config.ts`.
- Movement defaults (spec §5.1): walk 1.4 m/s, run 3.5 m/s (slope < 20°, stamina > 10), slide at slope 30–50°, climb 0.5 m/s at slope > 50°; run −8/s, climb moving −5/s, climb holding −1.5/s; fall < 6 m none, 6–15 m injury (−30 stamina cap), > 15 m fatal.
- Rock is climbable without gear; snow needs an ice axe; ice needs ice axe + crampons. M1 has no gear.
- Assets must be CC0 (or Mixamo terms) and listed in `CREDITS.md`.
- DEM attribution string, verbatim: `Contains modified Copernicus DEM data`.
- Pin `typescript@~5.9` (TypeScript 7 is the native compiler and is not yet supported by typescript-eslint).
- Run `npm run format` before every commit. When Claude creates a commit, end the message with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Deliberate deviations from the spec (recorded in the spec by Task 16)

1. Heightmap tiles are raw little-endian `uint16` (`.u16`, 257 × 257 samples with shared edges), not 16-bit PNG — browsers cannot decode 16-bit PNG losslessly through canvas.
2. Splat maps are deferred to M2; M1 classifies rock/snow/ice at runtime from slope and altitude.
3. Camp positions are generated by the pipeline into the terrain `manifest.json`, not hand-written into the mountain JSON.
4. Camera collision samples the terrain heightfield instead of a physics sphere-cast.
5. M1 uses the CC0 "RobotExpressive" model as a placeholder character (no climb animation).
6. Baked terrain is generated, not committed. CI uses a small synthetic `test-peak`.

`tools/terrain/terrain_lib.py`:

```python
"""Pure helpers for the Mountain Odyssey terrain pipeline (spec section 6.5).

World axes: x = east, z = south, y = up, all in metres. Grid sample (row r, col c)
sits at x = origin + c * spacing, z = origin + r * spacing. Row 0 is the north edge.
"""
from __future__ import annotations

import json
import math
from dataclasses import dataclass
from pathlib import Path

import numpy as np
from PIL import Image

M_PER_DEG_LAT = 111_320.0


@dataclass(frozen=True)
class GridSpec:
    center_lat: float
    center_lon: float
    tiles: int
    tile_cells: int
    spacing_m: float

    @property
    def samples(self) -> int:
        return self.tiles * self.tile_cells + 1

    @property
    def size_m(self) -> float:
        return self.tiles * self.tile_cells * self.spacing_m

    @property
    def origin(self) -> float:
        return -self.size_m / 2.0


def m_per_deg_lon(lat: float) -> float:
    return M_PER_DEG_LAT * math.cos(math.radians(lat))


def latlon_bounds(spec: GridSpec, margin_m: float = 200.0) -> tuple[float, float, float, float]:
    """(west, south, east, north) in degrees covering the grid plus a margin."""
    half = spec.size_m / 2.0 + margin_m
    dlat = half / M_PER_DEG_LAT
    dlon = half / m_per_deg_lon(spec.center_lat)
    return (
        spec.center_lon - dlon,
        spec.center_lat - dlat,
        spec.center_lon + dlon,
        spec.center_lat + dlat,
    )


def bilinear_sample(grid: np.ndarray, rows: np.ndarray, cols: np.ndarray) -> np.ndarray:
    grid = np.asarray(grid, dtype=np.float64)
    r0 = np.clip(np.floor(rows).astype(np.int64), 0, grid.shape[0] - 2)
    c0 = np.clip(np.floor(cols).astype(np.int64), 0, grid.shape[1] - 2)
    fr = np.clip(rows - r0, 0.0, 1.0)
    fc = np.clip(cols - c0, 0.0, 1.0)
    top = grid[r0, c0] * (1 - fc) + grid[r0, c0 + 1] * fc
    bottom = grid[r0 + 1, c0] * (1 - fc) + grid[r0 + 1, c0 + 1] * fc
    return top * (1 - fr) + bottom * fr


def resample_to_metric(
    dem: np.ndarray,
    dem_bounds: tuple[float, float, float, float],
    spec: GridSpec,
    coarse_spacing_m: float,
) -> np.ndarray:
    """Resample a lat/lon DEM (row 0 = north) onto the local metric grid.

    dem_bounds are the (west, south, east, north) coordinates of the DEM's outer
    pixel centres.
    """
    west, south, east, north = dem_bounds
    n = int(round(spec.size_m / coarse_spacing_m)) + 1
    offsets = spec.origin + np.arange(n) * coarse_spacing_m
    lat = spec.center_lat - offsets / M_PER_DEG_LAT
    lon = spec.center_lon + offsets / m_per_deg_lon(spec.center_lat)
    rows = (north - lat) / (north - south) * (dem.shape[0] - 1)
    cols = (lon - west) / (east - west) * (dem.shape[1] - 1)
    rr, cc = np.meshgrid(rows, cols, indexing="ij")
    return bilinear_sample(dem, rr, cc).astype(np.float32)


def upsample(h: np.ndarray, out_samples: int) -> np.ndarray:
    img = Image.fromarray(np.ascontiguousarray(h, dtype=np.float32))
    resized = img.resize((out_samples, out_samples), Image.Resampling.BICUBIC)
    return np.asarray(resized, dtype=np.float32)


def detail_noise(
    n: int,
    spacing_m: float,
    seed: int,
    octaves: tuple[tuple[float, float], ...] = ((64.0, 1.2), (16.0, 0.5), (6.0, 0.2)),
) -> np.ndarray:
    """Smooth value noise: sum of (wavelength_m, amplitude_m) octaves."""
    rng = np.random.default_rng(seed)
    out = np.zeros((n, n), dtype=np.float32)
    extent = (n - 1) * spacing_m
    for wavelength, amplitude in octaves:
        cells = max(2, int(math.ceil(extent / wavelength)) + 1)
        coarse = rng.uniform(-1.0, 1.0, size=(cells, cells)).astype(np.float32)
        out += upsample(coarse, n) * amplitude
    return out


def split_tiles(h: np.ndarray, tile_cells: int) -> dict[tuple[int, int], np.ndarray]:
    """Split into (tile_cells + 1)^2 tiles keyed (tx, tz); neighbours share edges."""
    tiles = (h.shape[0] - 1) // tile_cells
    out: dict[tuple[int, int], np.ndarray] = {}
    for tz in range(tiles):
        for tx in range(tiles):
            r0, c0 = tz * tile_cells, tx * tile_cells
            out[(tx, tz)] = h[r0 : r0 + tile_cells + 1, c0 : c0 + tile_cells + 1]
    return out


def quantize(h: np.ndarray, min_h: float, max_h: float) -> np.ndarray:
    q = np.round((np.asarray(h, dtype=np.float64) - min_h) / (max_h - min_h) * 65535.0)
    return np.clip(q, 0, 65535).astype("<u2")


def downsample(h: np.ndarray, step: int) -> np.ndarray:
    return np.ascontiguousarray(h[::step, ::step])


def slope_deg(h: np.ndarray, spacing_m: float) -> np.ndarray:
    gz, gx = np.gradient(np.asarray(h, dtype=np.float64), spacing_m)
    return np.degrees(np.arctan(np.hypot(gx, gz)))


def find_summit(h: np.ndarray) -> tuple[int, int]:
    r, c = np.unravel_index(int(np.argmax(h)), h.shape)
    return int(r), int(c)


def find_camp(
    h: np.ndarray,
    spacing_m: float,
    summit_rc: tuple[int, int],
    bearing_deg: float,
    target_elev_m: float,
    search_radius_m: float = 150.0,
    margin_cells: int = 8,
) -> tuple[int, int]:
    """Walk from the summit along a compass bearing (0 = north, 90 = east) until the
    terrain drops to target_elev_m, then pick the flattest nearby cell."""
    n = h.shape[0]
    dr = -math.cos(math.radians(bearing_deg))
    dc = math.sin(math.radians(bearing_deg))
    r, c = float(summit_rc[0]), float(summit_rc[1])
    while True:
        nr, nc = r + dr, c + dc
        inside = margin_cells <= nr < n - margin_cells and margin_cells <= nc < n - margin_cells
        if not inside:
            break
        r, c = nr, nc
        if h[int(round(r)), int(round(c))] <= target_elev_m:
            break
    ri, ci = int(round(r)), int(round(c))
    rad = max(1, int(search_radius_m / spacing_m))
    r_lo, r_hi = max(0, ri - rad), min(n, ri + rad + 1)
    c_lo, c_hi = max(0, ci - rad), min(n, ci + rad + 1)
    window = h[r_lo:r_hi, c_lo:c_hi]
    score = slope_deg(window, spacing_m) + np.abs(window - target_elev_m) / 20.0
    rr, cc = np.mgrid[r_lo:r_hi, c_lo:c_hi]
    score[np.hypot(rr - ri, cc - ci) > rad] = np.inf
    br, bc = np.unravel_index(int(np.argmin(score)), score.shape)
    return int(r_lo + br), int(c_lo + bc)


def cell_to_world(rc: tuple[int, int], origin_m: float, spacing_m: float, h: np.ndarray) -> list[float]:
    r, c = rc
    return [
        round(origin_m + c * spacing_m, 2),
        round(float(h[r, c]), 2),
        round(origin_m + r * spacing_m, 2),
    ]


def synthetic_heightmap(samples: int, spacing_m: float) -> np.ndarray:
    """Test mountain: 5000 m plain, a ~19 degree cone rising 240 m within 700 m of
    the centre, and an 8 m cliff band at radius 300 m."""
    half = (samples - 1) * spacing_m / 2.0
    coords = np.arange(samples) * spacing_m - half
    x, z = np.meshgrid(coords, coords)
    r = np.hypot(x, z)
    cone = np.clip(1.0 - r / 700.0, 0.0, 1.0) * 240.0
    cliff = 8.0 * np.clip((300.0 - r) / 1.0 + 0.5, 0.0, 1.0)
    return (5000.0 + cone + cliff).astype(np.float32)


def write_terrain(
    out_dir: str | Path,
    mountain_id: str,
    h: np.ndarray,
    spacing_m: float,
    tile_cells: int,
    origin_m: float,
    camps: list[dict],
    attribution: str,
    backdrop_spacing_m: float,
) -> dict:
    out = Path(out_dir)
    tiles_dir = out / "tiles"
    tiles_dir.mkdir(parents=True, exist_ok=True)
    min_h = float(math.floor(float(h.min()))) - 1.0
    max_h = float(math.ceil(float(h.max()))) + 1.0
    for (tx, tz), tile in split_tiles(h, tile_cells).items():
        (tiles_dir / f"{tx}_{tz}.u16").write_bytes(quantize(tile, min_h, max_h).tobytes())
    step = int(round(backdrop_spacing_m / spacing_m))
    if (h.shape[0] - 1) % step != 0:
        raise ValueError("backdrop spacing must divide the grid size")
    backdrop = downsample(h, step)
    (out / "backdrop.u16").write_bytes(quantize(backdrop, min_h, max_h).tobytes())
    tiles = (h.shape[0] - 1) // tile_cells
    manifest = {
        "id": mountain_id,
        "spacingM": spacing_m,
        "tileSamples": tile_cells + 1,
        "tilesX": tiles,
        "tilesZ": tiles,
        "originX": origin_m,
        "originZ": origin_m,
        "minHeightM": min_h,
        "maxHeightM": max_h,
        "backdrop": {"file": "backdrop.u16", "samples": int(backdrop.shape[0]), "spacingM": backdrop_spacing_m},
        "camps": camps,
        "attribution": attribution,
    }
    (out / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
    return manifest
```

```python
"""Bake Mountain Odyssey terrain tiles.

  python3 tools/terrain/bake.py --synthetic --out public/terrain/test-peak
  python3 tools/terrain/bake.py --config tools/terrain/mera-peak.config.json --out public/terrain/mera-peak

import json
import sys

    GridSpec,
    cell_to_world,
    detail_noise,
    find_camp,
    find_summit,
    latlon_bounds,
    resample_to_metric,
    synthetic_heightmap,
    upsample,
    write_terrain,

def bake_synthetic(out: str) -> dict:
    spec = GridSpec(0.0, 0.0, tiles=4, tile_cells=256, spacing_m=2.0)
    h = synthetic_heightmap(spec.samples, spec.spacing_m)

    def camp(camp_id: str, name: str, x: float, z: float) -> dict:
        rc = (int(round((z - spec.origin) / spec.spacing_m)), int(round((x - spec.origin) / spec.spacing_m)))
        return {"id": camp_id, "name": name, "position": cell_to_world(rc, spec.origin, spec.spacing_m, h)}

    camps = [camp("khare", "Khare", 0.0, 800.0), camp("high-camp", "High Camp", 0.0, 0.0)]
    return write_terrain(out, "test-peak", h, spec.spacing_m, spec.tile_cells, spec.origin, camps,
                         "Synthetic test terrain", backdrop_spacing_m=64.0)


def load_dem(url: str, spec: GridSpec):
    import rasterio
    from rasterio.windows import from_bounds

    west, south, east, north = latlon_bounds(spec)
    with rasterio.Env(GDAL_DISABLE_READDIR_ON_OPEN="EMPTY_DIR", AWS_NO_SIGN_REQUEST="YES"):
        with rasterio.open(url) as src:
            window = from_bounds(west, south, east, north, transform=src.transform)
            window = window.round_offsets().round_lengths()
            dem = src.read(1, window=window).astype(np.float32)
            t = src.window_transform(window)
    west_c = t.c + t.a / 2
    north_c = t.f + t.e / 2
    east_c = west_c + (dem.shape[1] - 1) * t.a
    south_c = north_c + (dem.shape[0] - 1) * t.e
    return dem, (west_c, south_c, east_c, north_c)


def bake_dem(config_path: str, out: str) -> dict:
    with open(config_path) as f:
        cfg = json.load(f)
    spec = GridSpec(cfg["centerLat"], cfg["centerLon"], cfg["tiles"], cfg["tileCells"], cfg["spacingM"])
    print(f"Reading DEM window for {cfg['id']} ...", flush=True)
    dem, bounds = load_dem(cfg["demUrl"], spec)
    coarse = resample_to_metric(dem, bounds, spec, cfg["coarseSpacingM"])
    print(f"Upsampling to {spec.samples}x{spec.samples} at {spec.spacing_m} m ...", flush=True)
    h = upsample(coarse, spec.samples)
    h += detail_noise(spec.samples, spec.spacing_m, cfg["noiseSeed"])
    summit = find_summit(h)
    camps = []
    for c in cfg["camps"]:
        rc = find_camp(h, spec.spacing_m, summit, c["bearingFromSummitDeg"], c["targetElevationM"])
        camps.append({"id": c["id"], "name": c["name"], "position": cell_to_world(rc, spec.origin, spec.spacing_m, h)})
    camps.append({"id": "summit", "name": "Summit", "position": cell_to_world(summit, spec.origin, spec.spacing_m, h)})
    return write_terrain(out, cfg["id"], h, spec.spacing_m, spec.tile_cells, spec.origin, camps,
                         cfg["attribution"], cfg["backdropSpacingM"])


def main(argv: list[str]) -> int:
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    source = parser.add_mutually_exclusive_group(required=True)
    source.add_argument("--synthetic", action="store_true", help="bake the synthetic test-peak")
    source.add_argument("--config", help="path to a mountain bake config JSON")
    parser.add_argument("--out", required=True, help="output directory")
    args = parser.parse_args(argv)
    manifest = bake_synthetic(args.out) if args.synthetic else bake_dem(args.config, args.out)
    print(f"Wrote {manifest['tilesX'] * manifest['tilesZ']} tiles to {args.out}")
    for camp in manifest["camps"]:
        x, y, z = camp["position"]
        print(f"  {camp['id']:<10} x={x:>9.1f}  z={z:>9.1f}  elevation={y:>7.1f} m")
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```ts
type Handler<T> = (payload: T) => void;

/** Minimal typed pub/sub used for cross-system engine events. */
export class EventBus<E extends object> {
  private readonly handlers = new Map<keyof E, Set<Handler<never>>>();

  on<K extends keyof E>(type: K, handler: Handler<E[K]>): () => void {
    const set = this.handlers.get(type) ?? new Set<Handler<never>>();
    this.handlers.set(type, set);
    set.add(handler);
    return () => {
      set.delete(handler);
    };
  }

  emit<K extends keyof E>(type: K, payload: E[K]): void {
    const set = this.handlers.get(type);
    if (!set) return;
    for (const handler of [...set]) (handler as Handler<E[K]>)(payload);
  }

  clear(): void {
    this.handlers.clear();
  }
}
```

```ts
export interface LoopCallbacks {
  step(dt: number): void;
  render(alpha: number, frameDt: number): void;
}

/** Fixed-timestep simulation with interpolated rendering (spec §6.3). */
export class GameLoop {
  readonly stepDt: number;
  private accumulator = 0;
  private paused = false;
  private rafId: number | null = null;
  private lastTime = 0;

  constructor(
    private readonly callbacks: LoopCallbacks,
    stepHz = 60,
    private readonly maxStepsPerFrame = 5,
  ) {
    this.stepDt = 1 / stepHz;
  }

  get isPaused(): boolean {
    return this.paused;
  }

  setPaused(paused: boolean): void {
    this.paused = paused;
    if (paused) this.accumulator = 0;
  }

  /** Advances by one real frame. Returns the number of fixed steps run. */
  advance(frameDt: number): number {
    const dt = Math.min(Math.max(frameDt, 0), 0.25);
    let steps = 0;
    if (!this.paused) {
      this.accumulator += dt;
      while (this.accumulator >= this.stepDt && steps < this.maxStepsPerFrame) {
        this.callbacks.step(this.stepDt);
        this.accumulator -= this.stepDt;
        steps++;
      }
      if (this.accumulator >= this.stepDt) this.accumulator = 0;
    }
    this.callbacks.render(this.paused ? 1 : this.accumulator / this.stepDt, dt);
    return steps;
  }

  start(): void {
    if (this.rafId !== null) return;
    this.lastTime = performance.now();
    const tick = (now: number) => {
      this.rafId = requestAnimationFrame(tick);
      const frameDt = (now - this.lastTime) / 1000;
      this.lastTime = now;
      this.advance(frameDt);
    };
    this.rafId = requestAnimationFrame(tick);
  }

  stop(): void {
    if (this.rafId !== null) cancelAnimationFrame(this.rafId);
    this.rafId = null;
  }
}
```

```ts
import { CONTROLS, type Action } from '../config/controls.config';

const PREVENT_DEFAULT = new Set(['Space', 'Tab']);

export class InputManager {
  private readonly down = new Set<string>();
  private readonly pressed = new Set<string>();
  private mouseDx = 0;
  private mouseDy = 0;
  private wheel = 0;
  private target: HTMLElement | null = null;

  onKeyDown(code: string, repeat = false): void {
    if (!repeat && !this.down.has(code)) this.pressed.add(code);
    this.down.add(code);
  }

  onKeyUp(code: string): void {
    this.down.delete(code);
  }

  onMouseMove(dx: number, dy: number): void {
    this.mouseDx += dx;
    this.mouseDy += dy;
  }

  onWheel(dy: number): void {
    this.wheel += dy;
  }

  reset(): void {
    this.down.clear();
    this.pressed.clear();
    this.mouseDx = 0;
    this.mouseDy = 0;
    this.wheel = 0;
  }

  isDown(action: Action): boolean {
    return CONTROLS[action].some((code) => this.down.has(code));
  }

  /** True once per physical key press; the press is consumed. */
  consumePressed(action: Action): boolean {
    let hit = false;
    for (const code of CONTROLS[action]) if (this.pressed.delete(code)) hit = true;
    return hit;
  }

  consumeMouse(): { dx: number; dy: number; wheel: number } {
    const result = { dx: this.mouseDx, dy: this.mouseDy, wheel: this.wheel };
    this.mouseDx = 0;
    this.mouseDy = 0;
    this.wheel = 0;
    return result;
  }

  axis(): { forward: number; right: number } {
    const v = (a: Action) => (this.isDown(a) ? 1 : 0);
    return { forward: v('forward') - v('back'), right: v('right') - v('left') };
  }

  attach(target: HTMLElement): void {
    this.target = target;
    window.addEventListener('keydown', this.handleKeyDown);
    window.addEventListener('keyup', this.handleKeyUp);
    window.addEventListener('blur', this.handleBlur);
    document.addEventListener('mousemove', this.handleMouseMove);
    target.addEventListener('wheel', this.handleWheel, { passive: true });
  }

  detach(): void {
    window.removeEventListener('keydown', this.handleKeyDown);
    window.removeEventListener('keyup', this.handleKeyUp);
    window.removeEventListener('blur', this.handleBlur);
    document.removeEventListener('mousemove', this.handleMouseMove);
    this.target?.removeEventListener('wheel', this.handleWheel);
    this.target = null;
  }

  private readonly handleKeyDown = (e: KeyboardEvent): void => {
    if (PREVENT_DEFAULT.has(e.code)) e.preventDefault();
    this.onKeyDown(e.code, e.repeat);
  };

  private readonly handleKeyUp = (e: KeyboardEvent): void => this.onKeyUp(e.code);

  private readonly handleBlur = (): void => this.reset();

  private readonly handleMouseMove = (e: MouseEvent): void => {
    if (this.target && document.pointerLockElement === this.target) {
      this.onMouseMove(e.movementX, e.movementY);
    }
  };

  private readonly handleWheel = (e: WheelEvent): void => this.onWheel(e.deltaY);
}
```

```ts
export type Surface = 'rock' | 'snow' | 'ice';

export interface SurfaceRules {
  rockMinSlopeDeg: number;
  snowlineM: number;
  iceMinAltitudeM: number;
  iceMinSlopeDeg: number;
}

export interface Gear {
  iceAxe: boolean;
  crampons: boolean;
}

export const NO_GEAR: Gear = { iceAxe: false, crampons: false };

/** M1 runtime classification; baked splat maps replace this in M2. */
export function surfaceAt(slopeDeg: number, altitudeM: number, rules: SurfaceRules): Surface {
  if (slopeDeg >= rules.rockMinSlopeDeg) return 'rock';
  if (altitudeM < rules.snowlineM) return 'rock';
  if (altitudeM >= rules.iceMinAltitudeM && slopeDeg >= rules.iceMinSlopeDeg) return 'ice';
  return 'snow';
}

/** Null when climbable with this gear, otherwise the HUD message (spec §5.1). */
export function climbRequirement(surface: Surface, gear: Gear): string | null {
  if (surface === 'snow') return gear.iceAxe ? null : 'Requires ice axe';
  if (surface === 'ice') return gear.iceAxe && gear.crampons ? null : 'Requires ice axe + crampons';
  return null;
}

/** sRGB colours, 0–1. */
export const SURFACE_COLORS: Record<Surface, readonly [number, number, number]> = {
  rock: [0.43, 0.41, 0.39],
  snow: [0.93, 0.95, 0.97],
  ice: [0.72, 0.83, 0.9],
};
```

```ts
export interface CampDef {
  id: string;
  name: string;
  position: [number, number, number];
}

export interface TerrainManifest {
  id: string;
  spacingM: number;
  tileSamples: number;
  tilesX: number;
  tilesZ: number;
  originX: number;
  originZ: number;
  minHeightM: number;
  maxHeightM: number;
  backdrop: { file: string; samples: number; spacingM: number };
  camps: CampDef[];
  attribution: string;
}

/** Heights in metres, row-major with row = z (south). */
export interface TileData {
  tx: number;
  tz: number;
  heights: Float32Array;
}

export interface Backdrop {
  samples: number;
  spacingM: number;
  heights: Float32Array;
}

const REQUIRED: (keyof TerrainManifest)[] = [
  'id',
  'spacingM',
  'tileSamples',
  'tilesX',
  'tilesZ',
  'originX',
  'originZ',
  'minHeightM',
  'maxHeightM',
  'backdrop',
  'camps',
];

export function decodeHeights(buffer: ArrayBuffer, minH: number, maxH: number): Float32Array {
  const raw = new Uint16Array(buffer); // tiles are little-endian; all supported browsers are LE
  const out = new Float32Array(raw.length);
  const k = (maxH - minH) / 65535;
  for (let i = 0; i < raw.length; i++) out[i] = minH + raw[i] * k;
  return out;
}

export const tileKey = (tx: number, tz: number): string => `${tx}_${tz}`;

export const tileUrl = (baseUrl: string, tx: number, tz: number): string =>
  `${baseUrl}/tiles/${tx}_${tz}.u16`;

export function parseManifest(json: unknown): TerrainManifest {
  const m = json as Partial<TerrainManifest>;
  for (const key of REQUIRED) {
    if (m[key] === undefined) throw new Error(`Terrain manifest is missing "${key}"`);
  }
  if (!Array.isArray(m.camps) || m.camps.length < 2) {
    throw new Error('Terrain manifest needs at least two camps');
  }
  return { attribution: '', ...m } as TerrainManifest;
}

export async function fetchWithRetry(
  url: string,
  attempts = 3,
  baseDelayMs = 300,
  fetchFn: typeof fetch = fetch,
): Promise<Response> {
  let lastError: unknown = null;
  for (let i = 0; i < attempts; i++) {
    try {
      const res = await fetchFn(url);
      if (res.ok) return res;
      lastError = new Error(`HTTP ${res.status} for ${url}`);
    } catch (err) {
      lastError = err;
    }
    if (i < attempts - 1) await new Promise((r) => setTimeout(r, baseDelayMs * 2 ** i));
  }
  throw lastError instanceof Error ? lastError : new Error(`Failed to load ${url}`);
}

export async function loadManifest(url: string, fetchFn?: typeof fetch): Promise<TerrainManifest> {
  const res = await fetchWithRetry(url, 3, 300, fetchFn);
  return parseManifest(await res.json());
}

export async function loadTile(
  baseUrl: string,
  m: TerrainManifest,
  tx: number,
  tz: number,
  fetchFn?: typeof fetch,
): Promise<TileData> {
  const url = tileUrl(baseUrl, tx, tz);
  const res = await fetchWithRetry(url, 3, 300, fetchFn);
  const heights = decodeHeights(await res.arrayBuffer(), m.minHeightM, m.maxHeightM);
  if (heights.length !== m.tileSamples * m.tileSamples) {
    throw new Error(`Tile ${url} has ${heights.length} samples, expected ${m.tileSamples ** 2}`);
  }
  return { tx, tz, heights };
}

export async function loadBackdrop(
  baseUrl: string,
  m: TerrainManifest,
  fetchFn?: typeof fetch,
): Promise<Backdrop> {
  const url = `${baseUrl}/${m.backdrop.file}`;
  const res = await fetchWithRetry(url, 3, 300, fetchFn);
  const heights = decodeHeights(await res.arrayBuffer(), m.minHeightM, m.maxHeightM);
  if (heights.length !== m.backdrop.samples ** 2) {
    throw new Error(`Backdrop ${url} has ${heights.length} samples`);
  }
  return { samples: m.backdrop.samples, spacingM: m.backdrop.spacingM, heights };
}
```

```ts
import { clamp, normalize, radToDeg, type Vec3 } from '../core/math';
import { tileKey, type TerrainManifest, type TileData } from './terrainData';

/** CPU-side height queries over the currently loaded tiles. */
export class TerrainSampler {
  private readonly tiles = new Map<string, Float32Array>();
  private readonly cells: number;

  constructor(private readonly m: TerrainManifest) {
    this.cells = m.tileSamples - 1;
  }

  setTile(tile: TileData): void {
    this.tiles.set(tileKey(tile.tx, tile.tz), tile.heights);
  }

  removeTile(tx: number, tz: number): void {
    this.tiles.delete(tileKey(tx, tz));
  }

  hasTile(tx: number, tz: number): boolean {
    return this.tiles.has(tileKey(tx, tz));
  }

  heightAt(x: number, z: number): number | null {
    const { m, cells } = this;
    const fx = (x - m.originX) / m.spacingM;
    const fz = (z - m.originZ) / m.spacingM;
    if (fx < 0 || fz < 0 || fx > cells * m.tilesX || fz > cells * m.tilesZ) return null;
    const tx = Math.min(Math.floor(fx / cells), m.tilesX - 1);
    const tz = Math.min(Math.floor(fz / cells), m.tilesZ - 1);
    const h = this.tiles.get(tileKey(tx, tz));
    if (!h) return null;
    const lx = fx - tx * cells;
    const lz = fz - tz * cells;
    const x0 = Math.min(Math.floor(lx), cells - 1);
    const z0 = Math.min(Math.floor(lz), cells - 1);
    const ax = lx - x0;
    const az = lz - z0;
    const s = m.tileSamples;
    const h00 = h[z0 * s + x0];
    const h10 = h[z0 * s + x0 + 1];
    const h01 = h[(z0 + 1) * s + x0];
    const h11 = h[(z0 + 1) * s + x0 + 1];
    return (h00 * (1 - ax) + h10 * ax) * (1 - az) + (h01 * (1 - ax) + h11 * ax) * az;
  }

  normalAt(x: number, z: number): Vec3 | null {
    const e = this.m.spacingM;
    const left = this.heightAt(x - e, z);
    const right = this.heightAt(x + e, z);
    const north = this.heightAt(x, z - e);
    const south = this.heightAt(x, z + e);
    if (left === null || right === null || north === null || south === null) return null;
    return normalize({ x: left - right, y: 2 * e, z: north - south });
  }

  slopeDegAt(x: number, z: number): number | null {
    const n = this.normalAt(x, z);
    return n ? radToDeg(Math.acos(clamp(n.y, -1, 1))) : null;
  }
}
```

```ts
import { SURFACE_COLORS, surfaceAt, type Surface, type SurfaceRules } from './surface';

export interface GridGeometryArgs {
  heights: Float32Array;
  samplesX: number;
  samplesZ: number;
  step: number;
  spacingM: number;
  originX: number;
  originZ: number;
  skirtDepthM: number;
  surface: SurfaceRules;
  hole?: { minX: number; minZ: number; maxX: number; maxZ: number };
}

export interface GridGeometryData {
  positions: Float32Array;
  normals: Float32Array;
  colors: Float32Array;
  indices: Uint32Array;
}

const srgbToLinear = (c: number): number =>
  c <= 0.04045 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;

const toLinear = (rgb: readonly [number, number, number]): [number, number, number] => [
  srgbToLinear(rgb[0]),
  srgbToLinear(rgb[1]),
  srgbToLinear(rgb[2]),
];

const LINEAR_COLORS: Record<Surface, [number, number, number]> = {
  rock: toLinear(SURFACE_COLORS.rock),
  snow: toLinear(SURFACE_COLORS.snow),
  ice: toLinear(SURFACE_COLORS.ice),
};

/** Deterministic 0–1 hash for subtle per-vertex colour variation. */
function hash2(x: number, z: number): number {
  const s = Math.sin(x * 12.9898 + z * 78.233) * 43758.5453;
  return s - Math.floor(s);
}

/** Builds an indexed terrain grid at a LOD step, with optional skirts and a hole. */
export function buildGridGeometry(a: GridGeometryArgs): GridGeometryData {
  const { heights, samplesX, samplesZ, step, spacingM, originX, originZ, skirtDepthM, hole } = a;
  if (step < 1 || (samplesX - 1) % step !== 0 || (samplesZ - 1) % step !== 0) {
    throw new Error(`LOD step ${step} must divide ${samplesX - 1} x ${samplesZ - 1} cells`);
  }
  const nx = (samplesX - 1) / step + 1;
  const nz = (samplesZ - 1) / step + 1;
  const hasSkirt = skirtDepthM > 0;
  const vertexCount = nx * nz + (hasSkirt ? 2 * (nx + nz) : 0);
  const positions = new Float32Array(vertexCount * 3);
  const normals = new Float32Array(vertexCount * 3);
  const colors = new Float32Array(vertexCount * 3);

  const sample = (sx: number, sz: number): number => {
    const cx = sx < 0 ? 0 : sx >= samplesX ? samplesX - 1 : sx;
    const cz = sz < 0 ? 0 : sz >= samplesZ ? samplesZ - 1 : sz;
    return heights[cz * samplesX + cx];
  };

  const writeVertex = (v: number, sx: number, sz: number, yOffset: number): void => {
    const x = originX + sx * spacingM;
    const z = originZ + sz * spacingM;
    const y = sample(sx, sz);
    const o = v * 3;
    positions[o] = x;
    positions[o + 1] = y + yOffset;
    positions[o + 2] = z;
    const gx = sample(sx - step, sz) - sample(sx + step, sz);
    const gz = sample(sx, sz - step) - sample(sx, sz + step);
    const gy = 2 * step * spacingM;
    const len = Math.hypot(gx, gy, gz);
    normals[o] = gx / len;
    normals[o + 1] = gy / len;
    normals[o + 2] = gz / len;
    const slopeDeg = (Math.acos(gy / len) * 180) / Math.PI;
    const c = LINEAR_COLORS[surfaceAt(slopeDeg, y, a.surface)];
    const shade = 0.9 + 0.1 * hash2(x, z);
    colors[o] = c[0] * shade;
    colors[o + 1] = c[1] * shade;
    colors[o + 2] = c[2] * shade;
  };

  for (let iz = 0; iz < nz; iz++) {
    for (let ix = 0; ix < nx; ix++) writeVertex(iz * nx + ix, ix * step, iz * step, 0);
  }

  const maxIndices = (nx - 1) * (nz - 1) * 6 + (hasSkirt ? 2 * (nx - 1 + (nz - 1)) * 6 : 0);
  const indices = new Uint32Array(maxIndices);
  let k = 0;
  const cell = step * spacingM;
  for (let iz = 0; iz < nz - 1; iz++) {
    for (let ix = 0; ix < nx - 1; ix++) {
      if (hole) {
        const cx = originX + (ix + 0.5) * cell;
        const cz = originZ + (iz + 0.5) * cell;
        if (cx > hole.minX && cx < hole.maxX && cz > hole.minZ && cz < hole.maxZ) continue;
      }
      const v0 = iz * nx + ix;
      const v1 = v0 + 1;
      const v2 = v0 + nx;
      const v3 = v2 + 1;
      indices[k++] = v0;
      indices[k++] = v2;
      indices[k++] = v1;
      indices[k++] = v1;
      indices[k++] = v2;
      indices[k++] = v3;
    }
  }

  if (hasSkirt) {
    let v = nx * nz;
    const edges: number[][] = [
      Array.from({ length: nx }, (_, ix) => ix),
      Array.from({ length: nx }, (_, ix) => (nz - 1) * nx + ix),
      Array.from({ length: nz }, (_, iz) => iz * nx),
      Array.from({ length: nz }, (_, iz) => iz * nx + nx - 1),
    ];
    for (const edge of edges) {
      const first = v;
      for (const top of edge) {
        writeVertex(v++, (top % nx) * step, Math.floor(top / nx) * step, -skirtDepthM);
      }
      for (let i = 0; i < edge.length - 1; i++) {
        const t0 = edge[i];
        const t1 = edge[i + 1];
        const s0 = first + i;
        const s1 = first + i + 1;
        indices[k++] = t0;
        indices[k++] = s0;
        indices[k++] = t1;
        indices[k++] = t1;
        indices[k++] = s0;
        indices[k++] = s1;
      }
    }
  }

  return { positions, normals, colors, indices: indices.slice(0, k) };
}
```

```ts
import { buildGridGeometry, type GridGeometryArgs, type GridGeometryData } from './gridGeometry';

export interface MeshBuilder {
  build(args: GridGeometryArgs): Promise<GridGeometryData>;
  dispose(): void;
}

/** Builds on the calling thread. Used by tests and as a fallback. */
export class SyncMeshBuilder implements MeshBuilder {
  async build(args: GridGeometryArgs): Promise<GridGeometryData> {
    return buildGridGeometry(args);
  }
  dispose(): void {}
}

interface Pending {
  resolve: (data: GridGeometryData) => void;
  reject: (err: Error) => void;
}

/** Builds terrain meshes off the main thread (spec §6.5). */
export class WorkerMeshBuilder implements MeshBuilder {
  private readonly worker: Worker;
  private readonly pending = new Map<number, Pending>();
  private nextId = 1;

  constructor() {
    this.worker = new Worker(new URL('./meshBuilder.worker.ts', import.meta.url), {
      type: 'module',
    });
    this.worker.onmessage = (
      e: MessageEvent<{ id: number; data?: GridGeometryData; error?: string }>,
    ) => {
      const p = this.pending.get(e.data.id);
      if (!p) return;
      this.pending.delete(e.data.id);
      if (e.data.data) p.resolve(e.data.data);
      else p.reject(new Error(e.data.error ?? 'Mesh worker failed'));
    };
    this.worker.onerror = (e) => this.rejectAll(new Error(e.message || 'Mesh worker crashed'));
  }

  build(args: GridGeometryArgs): Promise<GridGeometryData> {
    const id = this.nextId++;
    return new Promise((resolve, reject) => {
      this.pending.set(id, { resolve, reject });
      this.worker.postMessage({ id, args });
    });
  }

  dispose(): void {
    this.worker.terminate();
    this.rejectAll(new Error('Mesh builder disposed'));
  }

  private rejectAll(err: Error): void {
    for (const p of this.pending.values()) p.reject(err);
    this.pending.clear();
  }
}
```

```ts
import RAPIER from '@dimforge/rapier3d-compat';
import type { Vec3 } from '../core/math';
import { tileKey, type TerrainManifest, type TileData } from '../world/terrainData';

export interface RayHit {
  point: Vec3;
  normal: Vec3;
  distance: number;
}

let initPromise: Promise<void> | null = null;

export function initRapier(): Promise<void> {
  initPromise ??= RAPIER.init();
  return initPromise;
}

/** Thin wrapper over a Rapier world. Rapier simulates; it never renders. */
export class PhysicsWorld {
  readonly R = RAPIER;
  readonly world: RAPIER.World;
  private readonly terrain = new Map<string, RAPIER.Collider>();

  private constructor(gravityY: number) {
    this.world = new RAPIER.World({ x: 0, y: gravityY, z: 0 });
  }

  static async create(gravityY: number): Promise<PhysicsWorld> {
    await initRapier();
    return new PhysicsWorld(gravityY);
  }

  addTerrainTile(tile: TileData, m: TerrainManifest): void {
    const key = tileKey(tile.tx, tile.tz);
    if (this.terrain.has(key)) return;
    const s = m.tileSamples;
    const cells = s - 1;
    // Rapier heightfields are column-major: rows run along z, columns along x.
    const colMajor = new Float32Array(s * s);
    for (let z = 0; z < s; z++) {
      for (let x = 0; x < s; x++) colMajor[z + x * s] = tile.heights[z * s + x];
    }
    const size = cells * m.spacingM;
    const desc = RAPIER.ColliderDesc.heightfield(
      cells,
      cells,
      colMajor,
      { x: size, y: 1, z: size },
      RAPIER.HeightFieldFlags.FIX_INTERNAL_EDGES,
    ).setTranslation(m.originX + tile.tx * size + size / 2, 0, m.originZ + tile.tz * size + size / 2);
    this.terrain.set(key, this.world.createCollider(desc));
  }

  removeTerrainTile(key: string): void {
    const collider = this.terrain.get(key);
    if (!collider) return;
    this.world.removeCollider(collider, false);
    this.terrain.delete(key);
  }

  hasTerrainTile(key: string): boolean {
    return this.terrain.has(key);
  }

  terrainTileKeys(): string[] {
    return [...this.terrain.keys()];
  }

  step(dt: number): void {
    this.world.timestep = dt;
    this.world.step();
  }

  castRay(origin: Vec3, dir: Vec3, maxDist: number, exclude?: RAPIER.Collider): RayHit | null {
    const ray = new RAPIER.Ray(origin, dir);
    const hit = this.world.castRayAndGetNormal(ray, maxDist, true, undefined, undefined, exclude);
    if (!hit) return null;
    const p = ray.pointAt(hit.timeOfImpact);
    return {
      point: { x: p.x, y: p.y, z: p.z },
      normal: { x: hit.normal.x, y: hit.normal.y, z: hit.normal.z },
      distance: hit.timeOfImpact,
    };
  }

  dispose(): void {
    this.terrain.clear();
    this.world.free();
  }
}
```

```ts
import * as THREE from 'three';
import type { PhysicsWorld } from '../physics/PhysicsWorld';
import type { GridGeometryData } from './gridGeometry';
import type { MeshBuilder } from './meshBuilder';
import type { SurfaceRules } from './surface';
import {
  loadBackdrop,
  loadTile,
  tileKey,
  type Backdrop,
  type TerrainManifest,
  type TileData,
} from './terrainData';
import type { TerrainSampler } from './TerrainSampler';

export interface LodRing {
  maxRing: number;
  step: number;
}

export interface DesiredChunk {
  tx: number;
  tz: number;
  step: number;
}

interface TileIndex {
  tx: number;
  tz: number;
}

export function desiredChunks(
  cx: number,
  cz: number,
  tilesX: number,
  tilesZ: number,
  rings: readonly LodRing[],
): DesiredChunk[] {
  const radius = rings[rings.length - 1].maxRing;
  const out: DesiredChunk[] = [];
  for (let dz = -radius; dz <= radius; dz++) {
    for (let dx = -radius; dx <= radius; dx++) {
      const tx = cx + dx;
      const tz = cz + dz;
      if (tx < 0 || tz < 0 || tx >= tilesX || tz >= tilesZ) continue;
      const ring = Math.max(Math.abs(dx), Math.abs(dz));
      const lod = rings.find((r) => ring <= r.maxRing) ?? rings[rings.length - 1];
      out.push({ tx, tz, step: lod.step });
    }
  }
  return out;
}

export function colliderChunks(cx: number, cz: number, tilesX: number, tilesZ: number): TileIndex[] {
  return desiredChunks(cx, cz, tilesX, tilesZ, [{ maxRing: 1, step: 1 }]).map(({ tx, tz }) => ({
    tx,
    tz,
  }));
}

export function chunkBounds(
  chunks: readonly TileIndex[],
  m: Pick<TerrainManifest, 'originX' | 'originZ' | 'tileSamples' | 'spacingM'>,
): { minX: number; minZ: number; maxX: number; maxZ: number } {
  const size = (m.tileSamples - 1) * m.spacingM;
  const xs = chunks.map((c) => c.tx);
  const zs = chunks.map((c) => c.tz);
  return {
    minX: m.originX + Math.min(...xs) * size,
    minZ: m.originZ + Math.min(...zs) * size,
    maxX: m.originX + (Math.max(...xs) + 1) * size,
    maxZ: m.originZ + (Math.max(...zs) + 1) * size,
  };
}

export interface ChunkManagerDeps {
  manifest: TerrainManifest;
  baseUrl: string;
  parent: THREE.Object3D;
  physics: PhysicsWorld;
  sampler: TerrainSampler;
  builder: MeshBuilder;
  material: THREE.Material;
  rings: readonly LodRing[];
  skirtDepthM: number;
  surface: SurfaceRules;
  fetchFn?: typeof fetch;
}

/** Streams terrain tiles around the player: data, LOD meshes, colliders and backdrop. */
export class ChunkManager {
  private readonly tileData = new Map<string, TileData>();
  private readonly meshes = new Map<string, { mesh: THREE.Mesh; step: number }>();
  private backdrop: Backdrop | null = null;
  private backdropMesh: THREE.Mesh | null = null;
  private center: TileIndex | null = null;
  private target: TileIndex | null = null;
  private running: Promise<void> | null = null;
  private disposed = false;

  constructor(private readonly d: ChunkManagerDeps) {}

  async loadBackdrop(): Promise<void> {
    this.backdrop = await loadBackdrop(this.d.baseUrl, this.d.manifest, this.d.fetchFn);
  }

  /** Request terrain around a world position. Safe to call every frame. */
  update(x: number, z: number): Promise<void> {
    this.target = this.tileIndexAt(x, z);
    if (!this.running) {
      this.running = this.drain().finally(() => {
        this.running = null;
      });
    }
    return this.running;
  }

  dispose(): void {
    this.disposed = true;
    for (const key of [...this.meshes.keys()]) this.removeMesh(key);
    this.removeBackdropMesh();
    for (const key of this.d.physics.terrainTileKeys()) this.d.physics.removeTerrainTile(key);
    this.tileData.clear();
  }

  private tileIndexAt(x: number, z: number): TileIndex {
    const m = this.d.manifest;
    const size = (m.tileSamples - 1) * m.spacingM;
    const clampIndex = (v: number, n: number) => Math.min(Math.max(Math.floor(v), 0), n - 1);
    return {
      tx: clampIndex((x - m.originX) / size, m.tilesX),
      tz: clampIndex((z - m.originZ) / size, m.tilesZ),
    };
  }

  private async drain(): Promise<void> {
    while (!this.disposed && this.target && !sameTile(this.target, this.center)) {
      const next = this.target;
      await this.reconcile(next);
      this.center = next;
    }
  }

  private async reconcile(c: TileIndex): Promise<void> {
    const { manifest: m, physics, sampler } = this.d;
    const desired = desiredChunks(c.tx, c.tz, m.tilesX, m.tilesZ, this.d.rings);
    const desiredKeys = new Set(desired.map((k) => tileKey(k.tx, k.tz)));

    await Promise.all(
      desired
        .filter((k) => !this.tileData.has(tileKey(k.tx, k.tz)))
        .map(async (k) => {
          const tile = await loadTile(this.d.baseUrl, m, k.tx, k.tz, this.d.fetchFn);
          this.tileData.set(tileKey(k.tx, k.tz), tile);
          sampler.setTile(tile);
        }),
    );
    if (this.disposed) return;

    const colliderKeys = new Set(colliderChunks(c.tx, c.tz, m.tilesX, m.tilesZ).map((k) => tileKey(k.tx, k.tz)));
    for (const key of colliderKeys) {
      const tile = this.tileData.get(key);
      if (tile && !physics.hasTerrainTile(key)) physics.addTerrainTile(tile, m);
    }
    for (const key of physics.terrainTileKeys()) {
      if (!colliderKeys.has(key)) physics.removeTerrainTile(key);
    }

    const size = (m.tileSamples - 1) * m.spacingM;
    await Promise.all(
      desired.map(async (k) => {
        const key = tileKey(k.tx, k.tz);
        const existing = this.meshes.get(key);
        if (existing && existing.step === k.step) return;
        const tile = this.tileData.get(key);
        if (!tile) return;
        const data = await this.d.builder.build({
          heights: tile.heights,
          samplesX: m.tileSamples,
          samplesZ: m.tileSamples,
          step: k.step,
          spacingM: m.spacingM,
          originX: m.originX + k.tx * size,
          originZ: m.originZ + k.tz * size,
          skirtDepthM: this.d.skirtDepthM,
          surface: this.d.surface,
        });
        if (this.disposed) return;
        this.removeMesh(key);
        this.meshes.set(key, { mesh: this.createMesh(data, key), step: k.step });
      }),
    );
    if (this.disposed) return;

    for (const key of [...this.meshes.keys()]) if (!desiredKeys.has(key)) this.removeMesh(key);
    for (const [key, tile] of [...this.tileData]) {
      if (desiredKeys.has(key)) continue;
      this.tileData.delete(key);
      sampler.removeTile(tile.tx, tile.tz);
    }

    await this.rebuildBackdrop(chunkBounds(desired, m));
  }

  private async rebuildBackdrop(hole: { minX: number; minZ: number; maxX: number; maxZ: number }) {
    const b = this.backdrop;
    if (!b) return;
    const m = this.d.manifest;
    const data = await this.d.builder.build({
      heights: b.heights,
      samplesX: b.samples,
      samplesZ: b.samples,
      step: 1,
      spacingM: b.spacingM,
      originX: m.originX,
      originZ: m.originZ,
      skirtDepthM: 0,
      surface: this.d.surface,
      hole,
    });
    if (this.disposed) return;
    this.removeBackdropMesh();
    this.backdropMesh = this.createMesh(data, 'backdrop');
  }

  private createMesh(data: GridGeometryData, name: string): THREE.Mesh {
    const geometry = new THREE.BufferGeometry();
    geometry.setAttribute('position', new THREE.BufferAttribute(data.positions, 3));
    geometry.setAttribute('normal', new THREE.BufferAttribute(data.normals, 3));
    geometry.setAttribute('color', new THREE.BufferAttribute(data.colors, 3));
    geometry.setIndex(new THREE.BufferAttribute(data.indices, 1));
    geometry.computeBoundingSphere();
    const mesh = new THREE.Mesh(geometry, this.d.material);
    mesh.name = `terrain-${name}`;
    mesh.receiveShadow = true;
    mesh.matrixAutoUpdate = false;
    this.d.parent.add(mesh);
    return mesh;
  }

  private removeMesh(key: string): void {
    const entry = this.meshes.get(key);
    if (!entry) return;
    entry.mesh.removeFromParent();
    entry.mesh.geometry.dispose();
    this.meshes.delete(key);
  }

  private removeBackdropMesh(): void {
    if (!this.backdropMesh) return;
    this.backdropMesh.removeFromParent();
    this.backdropMesh.geometry.dispose();
    this.backdropMesh = null;
  }
}

function sameTile(a: TileIndex, b: TileIndex | null): boolean {
  return b !== null && a.tx === b.tx && a.tz === b.tz;
}
```

```ts
import type { Surface } from '../world/surface';

export type MoveState = 'walk' | 'run' | 'slide' | 'climb' | 'fall';

export interface MoveInput {
  forward: number;
  right: number;
  run: boolean;
  /** True only on the step the key was pressed. */
  grab: boolean;
  /** True only on the step the key was pressed. */
  jump: boolean;
}

export const NO_INPUT: MoveInput = { forward: 0, right: 0, run: false, grab: false, jump: false };

export interface WallInfo {
  slopeDeg: number;
  surface: Surface;
  climbable: boolean;
}

export interface MoveContext {
  grounded: boolean;
  groundSlopeDeg: number;
  wall: WallInfo | null;
  stamina: number;
}

export interface MovementRules {
  maxWalkSlopeDeg: number;
  maxRunSlopeDeg: number;
  minClimbSlopeDeg: number;
  runMinStamina: number;
}

export const isMoving = (i: MoveInput): boolean => i.forward !== 0 || i.right !== 0;

/** Pure movement state transitions (spec §5.1). */
export function nextMoveState(
  current: MoveState,
  input: MoveInput,
  ctx: MoveContext,
  rules: MovementRules,
): MoveState {
  const wall = ctx.wall;
  const steepWall = wall !== null && wall.slopeDeg >= rules.minClimbSlopeDeg;
  const climbableWall = steepWall && wall !== null && wall.climbable;

  if (current === 'climb') {
    if (ctx.stamina <= 0 || input.jump) return 'fall';
    if (ctx.grounded && input.forward < 0) return 'walk';
    if (!steepWall) return 'walk';
    if (!climbableWall) return 'fall';
    return 'climb';
  }

  if (input.grab && climbableWall && ctx.stamina > 0) return 'climb';
  if (!ctx.grounded) return 'fall';
  if (ctx.groundSlopeDeg >= rules.maxWalkSlopeDeg) return 'slide';

  const staminaOk = current === 'run' ? ctx.stamina > 0 : ctx.stamina > rules.runMinStamina;
  if (input.run && isMoving(input) && staminaOk && ctx.groundSlopeDeg < rules.maxRunSlopeDeg) {
    return 'run';
  }
  return 'walk';
}
```

```ts
import { clamp } from '../core/math';
import type { MoveState } from './movementState';

export interface StaminaState {
  value: number;
  cap: number;
}

export interface StaminaRules {
  max: number;
  runDrainPerSec: number;
  climbMoveDrainPerSec: number;
  climbHoldDrainPerSec: number;
  recoveryPerSec: number;
  minCap: number;
}

export const fullStamina = (r: StaminaRules): StaminaState => ({ value: r.max, cap: r.max });

export function staminaRate(state: MoveState, moving: boolean, r: StaminaRules): number {
  switch (state) {
    case 'run':
      return moving ? -r.runDrainPerSec : r.recoveryPerSec;
    case 'climb':
      return moving ? -r.climbMoveDrainPerSec : -r.climbHoldDrainPerSec;
    case 'walk':
      return r.recoveryPerSec;
    default:
      return 0;
  }
}

export function updateStamina(
  s: StaminaState,
  state: MoveState,
  moving: boolean,
  dt: number,
  r: StaminaRules,
): StaminaState {
  return { value: clamp(s.value + staminaRate(state, moving, r) * dt, 0, s.cap), cap: s.cap };
}

export function applyInjury(s: StaminaState, penalty: number, r: StaminaRules): StaminaState {
  const cap = Math.max(r.minCap, s.cap - penalty);
  return { value: Math.min(s.value, cap), cap };
}
```

```ts
import type RAPIER from '@dimforge/rapier3d-compat';
import { GAME_CONFIG as cfg } from '../config/game.config';
import type { EventBus } from '../core/EventBus';
import type { GameEvents } from '../core/events';
import {
  add,
  clamp,
  cross,
  degToRad,
  length,
  lerp,
  normalize,
  radToDeg,
  scale,
  vec3,
  type Vec3,
} from '../core/math';
import type { PhysicsWorld } from '../physics/PhysicsWorld';
import { climbRequirement, surfaceAt, type Gear } from '../world/surface';
import type { TerrainSampler } from '../world/TerrainSampler';
import { classifyFall } from './fallDamage';
import {
  isMoving,
  nextMoveState,
  type MoveInput,
  type MoveState,
  type WallInfo,
} from './movementState';
import { applyInjury, fullStamina, updateStamina, type StaminaState } from './stamina';

const UP: Vec3 = { x: 0, y: 1, z: 0 };
const UNCONTROLLED: ReadonlySet<MoveState> = new Set<MoveState>(['fall', 'slide']);

interface WallSense extends WallInfo {
  normal: Vec3;
  distance: number;
  requirement: string | null;
}

export interface PlayerSnapshot {
  feet: Vec3;
  state: MoveState;
  moving: boolean;
  stamina: StaminaState;
  facingYaw: number;
  prompt: string | null;
  grounded: boolean;
}

/** Third-person climber driven by Rapier's kinematic character controller. */
export class Player {
  private readonly body: RAPIER.RigidBody;
  private readonly collider: RAPIER.Collider;
  private readonly controller: RAPIER.KinematicCharacterController;
  private readonly centerOffset = cfg.player.capsuleHalfHeight + cfg.player.capsuleRadius;
  private state: MoveState = 'walk';
  private velocity: Vec3 = vec3();
  private grounded = false;
  private stamina: StaminaState = fullStamina(cfg.stamina);
  private facing: Vec3 = vec3(0, 0, -1);
  private wallNormal: Vec3 | null = null;
  private fallPeakY = 0;
  private moving = false;
  private prompt: string | null = null;
  private prevCenter: Vec3;
  private currCenter: Vec3;

  constructor(
    private readonly physics: PhysicsWorld,
    private readonly sampler: TerrainSampler,
    private readonly events: EventBus<GameEvents>,
    spawnFeet: Vec3,
    private readonly gear: Gear,
  ) {
    const R = physics.R;
    const c = { x: spawnFeet.x, y: spawnFeet.y + this.centerOffset, z: spawnFeet.z };
    this.body = physics.world.createRigidBody(
      R.RigidBodyDesc.kinematicPositionBased().setTranslation(c.x, c.y, c.z),
    );
    this.collider = physics.world.createCollider(
      R.ColliderDesc.capsule(cfg.player.capsuleHalfHeight, cfg.player.capsuleRadius),
      this.body,
    );
    this.controller = physics.world.createCharacterController(0.02);
    this.controller.setApplyImpulsesToDynamicBodies(false);
    this.setClimbMode(false);
    this.prevCenter = c;
    this.currCenter = c;
  }

  get snapshot(): PlayerSnapshot {
    return {
      feet: this.feetOf(this.currCenter),
      state: this.state,
      moving: this.moving,
      stamina: this.stamina,
      facingYaw: Math.atan2(this.facing.x, this.facing.z),
      prompt: this.prompt,
      grounded: this.grounded,
    };
  }

  interpolatedFeet(alpha: number): Vec3 {
    return this.feetOf(lerp(this.prevCenter, this.currCenter, alpha));
  }

  /** Decide state and queue this step's movement. Call before physics.step(). */
  step(input: MoveInput, cameraYaw: number, dt: number): void {
    const center = this.currCenter;
    const prev = this.state;
    this.moving = isMoving(input);
    if (UNCONTROLLED.has(prev)) this.fallPeakY = Math.max(this.fallPeakY, center.y);

    const dir = this.moveDirection(input, cameraYaw);
    if (prev !== 'climb' && this.moving) this.facing = dir;

    const wall = this.senseWall(center, prev);
    const grounded = prev === 'climb' ? this.feetNearGround(center) : this.grounded;
    const groundSlopeDeg = grounded ? (this.sampler.slopeDegAt(center.x, center.z) ?? 0) : 0;
    const next = nextMoveState(
      prev,
      input,
      { grounded, groundSlopeDeg, wall, stamina: this.stamina.value },
      cfg.movement,
    );

    let mantle: Vec3 | null = null;
    if (prev !== next) {
      if (UNCONTROLLED.has(prev) && (next === 'walk' || next === 'run')) this.land(center.y);
      if (!UNCONTROLLED.has(prev) && UNCONTROLLED.has(next)) this.fallPeakY = center.y;
      if (next === 'climb') {
        this.setClimbMode(true);
        this.velocity = vec3();
      }
      if (prev === 'climb') {
        this.setClimbMode(false);
        this.wallNormal = null;
        if (next === 'walk' && !grounded) mantle = this.mantleTarget(center);
      }
    }
    this.state = next;

    if (mantle) {
      this.body.setNextKinematicTranslation(mantle);
      this.grounded = true;
      this.velocity = vec3();
    } else {
      const translation =
        next === 'climb' && wall
          ? this.climbTranslation(input, wall, dt)
          : this.groundTranslation(next, dir, input, dt);
      this.controller.computeColliderMovement(this.collider, translation);
      const mv = this.controller.computedMovement();
      this.grounded = this.controller.computedGrounded();
      if (next === 'fall') {
        this.velocity.x = mv.x / dt;
        this.velocity.z = mv.z / dt;
      }
      if (this.grounded && next !== 'slide' && this.velocity.y < 0) this.velocity.y = 0;
      this.body.setNextKinematicTranslation({
        x: center.x + mv.x,
        y: center.y + mv.y,
        z: center.z + mv.z,
      });
    }

    this.stamina = updateStamina(this.stamina, next, this.moving, dt, cfg.stamina);
    this.prompt = this.computePrompt(next, wall);
  }

  /** Record the simulated position. Call after physics.step(). */
  afterPhysics(): void {
    this.prevCenter = this.currCenter;
    const t = this.body.translation();
    this.currCenter = { x: t.x, y: t.y, z: t.z };
  }

  teleport(feet: Vec3): void {
    const c = { x: feet.x, y: feet.y + this.centerOffset, z: feet.z };
    this.body.setTranslation(c, true);
    this.body.setNextKinematicTranslation(c);
    this.prevCenter = c;
    this.currCenter = c;
    this.velocity = vec3();
    this.state = 'walk';
    this.grounded = false;
    this.wallNormal = null;
    this.setClimbMode(false);
    this.stamina = fullStamina(cfg.stamina);
  }

  dispose(): void {
    this.physics.world.removeCharacterController(this.controller);
    this.physics.world.removeRigidBody(this.body);
  }

  private feetOf(center: Vec3): Vec3 {
    return { x: center.x, y: center.y - this.centerOffset, z: center.z };
  }

  private feetNearGround(center: Vec3): boolean {
    const ground = this.sampler.heightAt(center.x, center.z);
    return ground !== null && center.y - this.centerOffset - ground < 0.15;
  }

  private setClimbMode(climbing: boolean): void {
    const c = this.controller;
    if (climbing) {
      c.setMaxSlopeClimbAngle(Math.PI / 2);
      c.setMinSlopeSlideAngle(Math.PI / 2);
      c.disableSnapToGround();
      c.disableAutostep();
    } else {
      c.setMaxSlopeClimbAngle(degToRad(cfg.movement.maxWalkSlopeDeg));
      c.setMinSlopeSlideAngle(degToRad(cfg.movement.maxWalkSlopeDeg));
      c.enableSnapToGround(0.3);
      c.enableAutostep(0.35, 0.2, false);
    }
  }

  private moveDirection(input: MoveInput, yaw: number): Vec3 {
    const d = {
      x: -Math.sin(yaw) * input.forward + Math.cos(yaw) * input.right,
      y: 0,
      z: -Math.cos(yaw) * input.forward - Math.sin(yaw) * input.right,
    };
    return length(d) > 0 ? normalize(d) : d;
  }

  private senseWall(center: Vec3, state: MoveState): WallSense | null {
    const dir = state === 'climb' && this.wallNormal ? scale(this.wallNormal, -1) : this.facing;
    const reach = cfg.player.capsuleRadius + cfg.movement.grabReachM;
    const hit = this.physics.castRay(center, dir, reach, this.collider);
    if (!hit) return null;
    const slopeDeg = radToDeg(Math.acos(clamp(hit.normal.y, -1, 1)));
    const surface = surfaceAt(slopeDeg, hit.point.y, cfg.surface);
    const requirement = climbRequirement(surface, this.gear);
    return {
      slopeDeg,
      surface,
      climbable: requirement === null,
      requirement,
      normal: hit.normal,
      distance: hit.distance,
    };
  }

  private groundTranslation(state: MoveState, dir: Vec3, input: MoveInput, dt: number): Vec3 {
    const m = cfg.movement;
    const g = cfg.physics.gravity;
    switch (state) {
      case 'walk':
      case 'run': {
        const speed = state === 'run' ? m.runSpeed : m.walkSpeed;
        this.velocity.x = dir.x * speed;
        this.velocity.z = dir.z * speed;
        if (input.jump && this.grounded) {
          this.velocity.y = m.jumpVelocity;
          this.controller.disableSnapToGround();
        } else if (this.grounded) {
          this.velocity.y = -1;
        } else {
          this.velocity.y += g * dt;
        }
        break;
      }
      case 'slide':
        this.velocity.x = dir.x * m.walkSpeed * m.airControl;
        this.velocity.z = dir.z * m.walkSpeed * m.airControl;
        this.velocity.y = Math.max(this.velocity.y + g * dt, -m.maxSlideSpeed);
        break;
      case 'fall':
        this.velocity.x += dir.x * m.walkSpeed * m.airControl * dt;
        this.velocity.z += dir.z * m.walkSpeed * m.airControl * dt;
        this.velocity.y = Math.max(this.velocity.y + g * dt, -m.maxFallSpeed);
        break;
      case 'climb':
        break;
    }
    return scale(this.velocity, dt);
  }

  private climbTranslation(input: MoveInput, wall: WallSense, dt: number): Vec3 {
    const n = wall.normal;
    this.wallNormal = n;
    const face = normalize({ x: -n.x, y: 0, z: -n.z });
    this.facing = face;
    const right = normalize(cross(face, UP));
    const up = normalize(cross(n, right));
    let t = add(scale(right, input.right), scale(up, input.forward));
    if (length(t) > 0) t = scale(normalize(t), cfg.movement.climbSpeed * dt);
    const hold = cfg.player.capsuleRadius + cfg.movement.climbHoldGapM;
    const correction = clamp(wall.distance - hold, -0.05, 0.05);
    this.velocity = vec3();
    return add(t, scale(n, -correction));
  }

  private mantleTarget(center: Vec3): Vec3 | null {
    const p = add(center, scale(this.facing, cfg.movement.mantleForwardM));
    const ground = this.sampler.heightAt(p.x, p.z);
    if (ground === null) return null;
    const feet = center.y - this.centerOffset;
    if (ground > feet + cfg.movement.mantleMaxRiseM || ground < feet - 1) return null;
    return { x: p.x, y: ground + this.centerOffset + 0.02, z: p.z };
  }

  private land(centerY: number): void {
    const dropM = this.fallPeakY - centerY;
    const outcome = classifyFall(dropM, cfg.fall);
    if (outcome === 'injury') {
      this.stamina = applyInjury(this.stamina, cfg.fall.injuryCapPenalty, cfg.stamina);
    }
    if (outcome !== 'none') this.events.emit('player:fell', { dropM, outcome });
    this.setClimbMode(false);
  }

  private computePrompt(state: MoveState, wall: WallSense | null): string | null {
    if (state === 'climb') return 'Space — Let go';
    if (!wall || wall.slopeDeg < cfg.movement.minClimbSlopeDeg) return null;
    return wall.requirement ?? 'E — Grab';
  }
}
```

```ts
import { lerp, type Vec3 } from '../core/math';

/** Camera offset from its target. yaw 0 = camera on +Z looking toward -Z. */
export function orbitOffset(yaw: number, pitch: number, distance: number): Vec3 {
  return {
    x: Math.sin(yaw) * Math.cos(pitch) * distance,
    y: Math.sin(pitch) * distance,
    z: Math.cos(yaw) * Math.cos(pitch) * distance,
  };
}

/** March from target to desired; stop before the first point under the terrain. */
export function resolveCameraPosition(
  target: Vec3,
  desired: Vec3,
  groundAt: (x: number, z: number) => number | null,
  padding: number,
  samples = 12,
): Vec3 {
  let safe = target;
  for (let i = 1; i <= samples; i++) {
    const p = lerp(target, desired, i / samples);
    const g = groundAt(p.x, p.z);
    if (g !== null && p.y < g + padding) break;
    safe = p;
  }
  const g = groundAt(safe.x, safe.z);
  return g !== null && safe.y < g + padding ? { ...safe, y: g + padding } : safe;
}

export const smoothFactor = (sharpness: number, dt: number): number => 1 - Math.exp(-sharpness * dt);
```

```ts
import type * as THREE from 'three';
import { GAME_CONFIG } from '../config/game.config';
import { add, clamp, degToRad, lerp, scale, type Vec3 } from '../core/math';
import { orbitOffset, resolveCameraPosition, smoothFactor } from './cameraMath';

const c = GAME_CONFIG.camera;

export class ThirdPersonCamera {
  yaw = 0;
  pitch = degToRad(c.defaultPitchDeg);
  private distance: number = c.defaultDistance;
  private position: Vec3 | null = null;

  constructor(
    private readonly camera: THREE.PerspectiveCamera,
    private readonly groundAt: (x: number, z: number) => number | null,
  ) {}

  applyMouse(dx: number, dy: number): void {
    this.yaw -= dx * c.mouseSensitivity;
    this.pitch = clamp(
      this.pitch + dy * c.mouseSensitivity,
      degToRad(c.minPitchDeg),
      degToRad(c.maxPitchDeg),
    );
  }

  applyWheel(delta: number): void {
    this.distance = clamp(this.distance + delta * c.wheelSensitivity, c.minDistance, c.maxDistance);
  }

  cycleDistance(): void {
    const next = c.distanceSteps.find((s) => s > this.distance + 0.01);
    this.distance = next ?? c.distanceSteps[0];
  }

  /** Jump straight to the target position on the next update (e.g. after respawn). */
  snap(): void {
    this.position = null;
  }

  update(feet: Vec3, climbing: boolean, dt: number): void {
    let target: Vec3 = { x: feet.x, y: feet.y + GAME_CONFIG.player.eyeHeight, z: feet.z };
    let distance = this.distance;
    if (climbing) {
      distance += c.climbExtraDistance;
      const right = { x: Math.cos(this.yaw), y: 0, z: -Math.sin(this.yaw) };
      target = add(target, scale(right, c.climbSideOffset));
    }
    const desired = add(target, orbitOffset(this.yaw, this.pitch, distance));
    const resolved = resolveCameraPosition(target, desired, this.groundAt, c.collisionPaddingM);
    let p = this.position ? lerp(this.position, resolved, smoothFactor(c.followSharpness, dt)) : resolved;
    const g = this.groundAt(p.x, p.z);
    if (g !== null && p.y < g + c.collisionPaddingM) p = { ...p, y: g + c.collisionPaddingM };
    this.position = p;
    this.camera.position.set(p.x, p.y, p.z);
    this.camera.lookAt(target.x, target.y, target.z);
  }
}
```

```ts
import type { EventBus } from '../core/EventBus';
import type { GameEvents } from '../core/events';
import type { Vec3 } from '../core/math';
import type { CampDef } from '../world/terrainData';

/** Detects arrival at camps; the latest one is the respawn point (spec §5.5). */
export class CheckpointSystem {
  private lastIndex = 0;
  private readonly reached = new Set<string>();

  constructor(
    private readonly camps: readonly CampDef[],
    private readonly radiusM: number,
    private readonly maxHeightDiffM: number,
    private readonly events: EventBus<GameEvents>,
  ) {
    if (camps.length === 0) throw new Error('CheckpointSystem needs at least one camp');
    this.reached.add(camps[0].id);
  }

  get last(): CampDef {
    return this.camps[this.lastIndex];
  }

  update(feet: Vec3): void {
    this.camps.forEach((camp, index) => {
      if (this.reached.has(camp.id)) return;
      const [x, y, z] = camp.position;
      const near = Math.hypot(feet.x - x, feet.z - z) <= this.radiusM;
      if (!near || Math.abs(feet.y - y) > this.maxHeightDiffM) return;
      this.reached.add(camp.id);
      this.lastIndex = index;
      this.events.emit('checkpoint:reached', { id: camp.id, name: camp.name });
    });
  }
}
```

```ts
import * as THREE from 'three';
import { GLTFLoader } from 'three/addons/loaders/GLTFLoader.js';
import type { Vec3 } from '../core/math';
import type { MoveState } from './movementState';

const CLIPS = { idle: 'Idle', walk: 'Walking', run: 'Running', fall: 'Jump', climb: 'Walking' };

/** Character model and animation. Falls back to a capsule if the model fails to load. */
export class PlayerVisual {
  private mixer: THREE.AnimationMixer | null = null;
  private readonly actions = new Map<string, THREE.AnimationAction>();
  private current: THREE.AnimationAction | null = null;
  private yaw = 0;

  private constructor(private readonly root: THREE.Group) {}

  static async load(parent: THREE.Object3D, url: string, heightM: number): Promise<PlayerVisual> {
    const root = new THREE.Group();
    parent.add(root);
    const visual = new PlayerVisual(root);
    try {
      const gltf = await new GLTFLoader().loadAsync(url);
      const model = gltf.scene;
      const box = new THREE.Box3().setFromObject(model);
      const k = heightM / (box.max.y - box.min.y);
      model.scale.setScalar(k);
      model.position.y = -box.min.y * k;
      model.traverse((o) => {
        if ((o as THREE.Mesh).isMesh) o.castShadow = true;
      });
      root.add(model);
      visual.mixer = new THREE.AnimationMixer(model);
      for (const clip of gltf.animations) visual.actions.set(clip.name, visual.mixer.clipAction(clip));
    } catch (err) {
      console.warn('Character model failed to load; using a capsule instead.', err);
      const capsule = new THREE.Mesh(
        new THREE.CapsuleGeometry(0.3, 1.1, 4, 12),
        new THREE.MeshStandardMaterial({ color: 0xd9534f }),
      );
      capsule.position.y = heightM / 2;
      capsule.castShadow = true;
      root.add(capsule);
    }
    return visual;
  }

  update(feet: Vec3, facingYaw: number, state: MoveState, moving: boolean, dt: number): void {
    this.root.position.set(feet.x, feet.y, feet.z);
    const diff = Math.atan2(Math.sin(facingYaw - this.yaw), Math.cos(facingYaw - this.yaw));
    this.yaw += diff * Math.min(1, dt * 12);
    this.root.rotation.y = this.yaw;
    this.play(this.clipFor(state, moving));
    this.mixer?.update(dt);
  }

  dispose(): void {
    this.mixer?.stopAllAction();
    this.root.removeFromParent();
  }

  private clipFor(state: MoveState, moving: boolean): string {
    if (state === 'fall' || state === 'slide') return CLIPS.fall;
    if (state === 'climb') return moving ? CLIPS.climb : CLIPS.idle;
    if (!moving) return CLIPS.idle;
    return state === 'run' ? CLIPS.run : CLIPS.walk;
  }

  private play(name: string): void {
    const next = this.actions.get(name);
    if (!next || next === this.current) return;
    next.reset().fadeIn(0.2).play();
    this.current?.fadeOut(0.2);
    this.current = next;
  }
}
```

```ts
import * as THREE from 'three';
import { ThirdPersonCamera } from '../camera/ThirdPersonCamera';
import { GAME_CONFIG as cfg } from '../config/game.config';
import { PhysicsWorld } from '../physics/PhysicsWorld';
import type { MoveInput } from '../player/movementState';
import { Player } from '../player/Player';
import { PlayerVisual } from '../player/PlayerVisual';
import { CheckpointSystem } from '../systems/CheckpointSystem';
import { ChunkManager } from '../world/ChunkManager';
import { Environment } from '../world/Environment';
import { SyncMeshBuilder, WorkerMeshBuilder, type MeshBuilder } from '../world/meshBuilder';
import { NO_GEAR } from '../world/surface';
import { loadManifest, type CampDef } from '../world/terrainData';
import { TerrainSampler } from '../world/TerrainSampler';
import { EventBus } from './EventBus';
import type { GameEvents } from './events';
import { GameLoop } from './GameLoop';
import type { HudState } from './hud';
import { InputManager } from './InputManager';
import type { Vec3 } from './math';

export interface GameOptions {
  canvas: HTMLCanvasElement;
  terrainBaseUrl: string;
  characterUrl: string;
  publish: (patch: Partial<HudState>) => void;
  useWorker?: boolean;
}

interface World {
  physics: PhysicsWorld;
  sampler: TerrainSampler;
  chunks: ChunkManager;
  player: Player;
  visual: PlayerVisual;
  cameraCtrl: ThirdPersonCamera;
  checkpoints: CheckpointSystem;
}

/** Composition root: owns the renderer, systems and loop. The only API React uses. */
export class Game {
  private readonly events = new EventBus<GameEvents>();
  private readonly input = new InputManager();
  private readonly renderer: THREE.WebGLRenderer;
  private readonly scene = new THREE.Scene();
  private readonly camera: THREE.PerspectiveCamera;
  private readonly environment: Environment;
  private readonly loop: GameLoop;
  private readonly resizeObserver: ResizeObserver;
  private cleanup: (() => void)[] = [];
  private w: World | null = null;
  private pendingRespawn = false;
  private respawning = false;
  private failed = false;
  private toast: { text: string; until: number } | null = null;
  private hudElapsedMs = 0;
  private hudFrames = 0;

  static async create(options: GameOptions): Promise<Game> {
    const game = new Game(options);
    try {
      await game.init();
    } catch (err) {
      game.dispose();
      throw err;
    }
    return game;
  }

  private constructor(private readonly o: GameOptions) {
    this.renderer = new THREE.WebGLRenderer({
      canvas: o.canvas,
      antialias: true,
      powerPreference: 'high-performance',
    });
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.renderer.shadowMap.enabled = true;
    this.renderer.shadowMap.type = THREE.PCFShadowMap;
    this.renderer.toneMapping = THREE.ACESFilmicToneMapping;
    this.renderer.outputColorSpace = THREE.SRGBColorSpace;
    this.camera = new THREE.PerspectiveCamera(cfg.camera.fovDeg, 1, cfg.camera.near, cfg.camera.far);
    this.environment = new Environment(this.scene, true);
    this.loop = new GameLoop(
      { step: (dt) => this.step(dt), render: (alpha, frameDt) => this.render(alpha, frameDt) },
      cfg.physics.stepHz,
      cfg.physics.maxStepsPerFrame,
    );
    this.resizeObserver = new ResizeObserver(() => this.resize());
    this.resizeObserver.observe(o.canvas);
    this.resize();
  }

  start(): void {
    this.loop.start();
    this.setPaused(true);
  }

  dispose(): void {
    this.loop.stop();
    this.resizeObserver.disconnect();
    for (const fn of this.cleanup.reverse()) fn();
    this.cleanup = [];
    this.events.clear();
    if (document.pointerLockElement === this.o.canvas) document.exitPointerLock();
    this.renderer.dispose();
    this.w = null;
  }

  private async init(): Promise<void> {
    const o = this.o;
    const manifest = await loadManifest(`${o.terrainBaseUrl}/manifest.json`);
    const physics = await PhysicsWorld.create(cfg.physics.gravity);
    this.cleanup.push(() => physics.dispose());
    const sampler = new TerrainSampler(manifest);
    const builder: MeshBuilder = o.useWorker === false ? new SyncMeshBuilder() : new WorkerMeshBuilder();
    const material = new THREE.MeshStandardMaterial({
      vertexColors: true,
      roughness: 0.95,
      metalness: 0,
      side: THREE.DoubleSide,
    });
    const chunks = new ChunkManager({
      manifest,
      baseUrl: o.terrainBaseUrl,
      parent: this.scene,
      physics,
      sampler,
      builder,
      material,
      rings: cfg.terrain.lodRings,
      skirtDepthM: cfg.terrain.skirtDepthM,
      surface: cfg.surface,
    });
    this.cleanup.push(() => {
      chunks.dispose();
      builder.dispose();
      material.dispose();
    });

    await chunks.loadBackdrop();
    const start = manifest.camps[0];
    await chunks.update(start.position[0], start.position[2]);

    const player = new Player(physics, sampler, this.events, this.spawnPoint(sampler, start), NO_GEAR);
    this.cleanup.push(() => player.dispose());
    physics.step(this.loop.stepDt);
    player.afterPhysics();

    const visual = await PlayerVisual.load(this.scene, o.characterUrl, cfg.player.heightM);
    this.cleanup.push(() => visual.dispose());

    this.w = {
      physics,
      sampler,
      chunks,
      player,
      visual,
      cameraCtrl: new ThirdPersonCamera(this.camera, (x, z) => sampler.heightAt(x, z)),
      checkpoints: new CheckpointSystem(
        manifest.camps,
        cfg.checkpoints.radiusM,
        cfg.checkpoints.maxHeightDiffM,
        this.events,
      ),
    };
    this.wireEvents();
    this.attachDom();
    o.publish({
      status: 'ready',
      paused: true,
      attribution: manifest.attribution,
      lastCheckpoint: start.name,
      altitudeM: Math.round(player.snapshot.feet.y),
    });
  }

  private wireEvents(): void {
    this.cleanup.push(
      this.events.on('checkpoint:reached', ({ name }) => {
        this.showToast(`Checkpoint reached: ${name}`);
        this.o.publish({ lastCheckpoint: name });
      }),
      this.events.on('player:fell', ({ outcome }) => {
        if (outcome === 'fatal') this.pendingRespawn = true;
        else this.showToast('Injured — maximum stamina reduced');
      }),
      this.events.on('player:respawned', ({ checkpointName }) =>
        this.showToast(`You fell. Back at ${checkpointName}.`),
      ),
    );
  }

  private attachDom(): void {
    const canvas = this.o.canvas;
    this.input.attach(canvas);
    const onClick = () => {
      if (document.pointerLockElement !== canvas) {
        Promise.resolve(canvas.requestPointerLock()).catch(() => undefined);
      }
    };
    const onLockChange = () => this.setPaused(document.pointerLockElement !== canvas);
    const onVisibility = () => {
      if (document.hidden) this.setPaused(true);
    };
    const onContextLost = (e: Event) => {
      e.preventDefault();
      this.setPaused(true);
    };
    canvas.addEventListener('click', onClick);
    canvas.addEventListener('webglcontextlost', onContextLost);
    document.addEventListener('pointerlockchange', onLockChange);
    document.addEventListener('visibilitychange', onVisibility);
    this.cleanup.push(() => {
      this.input.detach();
      canvas.removeEventListener('click', onClick);
      canvas.removeEventListener('webglcontextlost', onContextLost);
      document.removeEventListener('pointerlockchange', onLockChange);
      document.removeEventListener('visibilitychange', onVisibility);
    });
  }

  private step(dt: number): void {
    const w = this.w;
    if (!w || this.respawning) return;
    const axis = this.input.axis();
    const move: MoveInput = {
      forward: axis.forward,
      right: axis.right,
      run: this.input.isDown('run'),
      grab: this.input.consumePressed('grab'),
      jump: this.input.consumePressed('jump'),
    };
    if (this.input.consumePressed('cameraDistance')) w.cameraCtrl.cycleDistance();
    w.player.step(move, w.cameraCtrl.yaw, dt);
    w.physics.step(dt);
    w.player.afterPhysics();
    w.checkpoints.update(w.player.snapshot.feet);
    if (this.pendingRespawn) {
      this.pendingRespawn = false;
      void this.respawn();
    }
  }

  private render(alpha: number, frameDt: number): void {
    const w = this.w;
    if (!w) return;
    const mouse = this.input.consumeMouse();
    if (!this.loop.isPaused) {
      w.cameraCtrl.applyMouse(mouse.dx, mouse.dy);
      w.cameraCtrl.applyWheel(mouse.wheel);
    }
    const snap = w.player.snapshot;
    const feet = w.player.interpolatedFeet(alpha);
    w.visual.update(feet, snap.facingYaw, snap.state, snap.moving, this.loop.isPaused ? 0 : frameDt);
    w.cameraCtrl.update(feet, snap.state === 'climb', frameDt);
    this.environment.follow(feet);
    if (!this.respawning) w.chunks.update(feet.x, feet.z).catch((err: unknown) => this.fail(err));
    this.renderer.render(this.scene, this.camera);
    this.publishHud(frameDt);
  }

  private async respawn(): Promise<void> {
    const w = this.w;
    if (!w) return;
    const camp = w.checkpoints.last;
    this.respawning = true;
    try {
      await w.chunks.update(camp.position[0], camp.position[2]);
    } catch (err) {
      this.fail(err);
      return;
    } finally {
      this.respawning = false;
    }
    w.player.teleport(this.spawnPoint(w.sampler, camp));
    w.cameraCtrl.snap();
    this.events.emit('player:respawned', { checkpointId: camp.id, checkpointName: camp.name });
  }

  private spawnPoint(sampler: TerrainSampler, camp: CampDef): Vec3 {
    const [x, y, z] = camp.position;
    return { x, y: (sampler.heightAt(x, z) ?? y) + 0.05, z };
  }

  private publishHud(frameDt: number): void {
    const w = this.w;
    if (!w) return;
    this.hudFrames++;
    this.hudElapsedMs += frameDt * 1000;
    if (this.hudElapsedMs < cfg.hud.publishIntervalMs) return;
    const fps = Math.round(this.hudFrames / (this.hudElapsedMs / 1000));
    this.hudFrames = 0;
    this.hudElapsedMs = 0;
    if (this.toast && performance.now() > this.toast.until) this.toast = null;
    const s = w.player.snapshot;
    this.o.publish({
      altitudeM: Math.round(s.feet.y),
      stamina: Math.round(s.stamina.value),
      staminaCap: Math.round(s.stamina.cap),
      moveState: s.state,
      prompt: s.prompt,
      toast: this.toast?.text ?? null,
      fps,
    });
  }

  private showToast(text: string): void {
    this.toast = { text, until: performance.now() + cfg.hud.toastMs };
  }

  private setPaused(paused: boolean): void {
    this.loop.setPaused(paused);
    if (paused) this.input.reset();
    this.o.publish({ paused });
  }

  private fail(err: unknown): void {
    if (this.failed) return;
    this.failed = true;
    this.setPaused(true);
    this.o.publish({ status: 'error', error: err instanceof Error ? err.message : String(err) });
  }

  private resize(): void {
    const { clientWidth: width, clientHeight: height } = this.o.canvas;
    if (width === 0 || height === 0) return;
    this.renderer.setSize(width, height, false);
    this.camera.aspect = width / height;
    this.camera.updateProjectionMatrix();
  }
}
```

`src/app/GameView.tsx`:

```tsx
import { useEffect, useRef } from 'react';
import { useStore } from 'zustand';
import type { MountainDef } from '../data/mountains';
import { Game } from '../engine/core/Game';
import { initialHudState } from '../engine/core/hud';
import { gameStore } from '../state/gameStore';
import { Hud } from '../ui/Hud';

const CHARACTER_URL = '/models/RobotExpressive.glb';

export function GameView({ mountain }: { mountain: MountainDef }) {
  const containerRef = useRef<HTMLDivElement>(null);
  const status = useStore(gameStore, (s) => s.status);
  const error = useStore(gameStore, (s) => s.error);

  useEffect(() => {
    const container = containerRef.current;
    if (!container) return;
    const canvas = document.createElement('canvas');
    canvas.className = 'game-canvas';
    container.appendChild(canvas);
    gameStore.setState(initialHudState, true);

    let game: Game | null = null;
    let cancelled = false;
    Game.create({
      canvas,
      terrainBaseUrl: mountain.terrainBaseUrl,
      characterUrl: CHARACTER_URL,
      publish: (patch) => {
        if (!cancelled) gameStore.setState(patch);
      },
    })
      .then((g) => {
        if (cancelled) {
          g.dispose();
          return;
        }
        game = g;
        g.start();
      })
      .catch((err: unknown) => {
        if (!cancelled) {
          gameStore.setState({ status: 'error', error: err instanceof Error ? err.message : String(err) });
        }
      });

    return () => {
      cancelled = true;
      game?.dispose();
      canvas.remove();
    };
  }, [mountain]);

  return (
    <div className="game-root" ref={containerRef}>
      {status === 'loading' && <div className="overlay">Loading {mountain.facts.name}…</div>}
      {status === 'error' && (
        <div className="overlay overlay--error" role="alert">
          <p>{error}</p>
          <button type="button" onClick={() => window.location.reload()}>
            Retry
          </button>
        </div>
      )}
      {status === 'ready' && <Hud />}
    </div>
  );
}
```
